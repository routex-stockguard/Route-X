import React, { useState } from 'react';
import {
  Edit3,
  FilePlus2,
  Globe,
  PackagePlus,
  Plus,
  RotateCcw,
  Save,
  Trash2,
} from 'lucide-react';
import {
  CustomDetailItem,
  CustomDetailPlacement,
  formatCurrency,
  GENERATED_IMAGES,
  Product,
  SiteConfig,
} from '../types';

interface SiteSettingsViewProps {
  siteConfig: SiteConfig;
  products: Product[];
  returnReasons: string[];
  onSaveSiteConfig: (nextConfig: SiteConfig, label: string) => void;
  onResetSiteConfig: () => void;
  onAddProduct: (product: Product, onSuccess: () => void) => string | null;
  onDeleteProduct: (sku: string) => void;
  onUpdateReturnReasons: (nextReasons: string[], label: string) => void;
}

export function SiteSettingsView({
  siteConfig,
  products,
  returnReasons,
  onSaveSiteConfig,
  onResetSiteConfig,
  onAddProduct,
  onDeleteProduct,
  onUpdateReturnReasons,
}: SiteSettingsViewProps) {
  const [draft, setDraft] = useState<SiteConfig>(siteConfig);

  const [detailTitle, setDetailTitle] = useState('');
  const [detailContent, setDetailContent] = useState('');
  const [detailPlacement, setDetailPlacement] =
    useState<CustomDetailPlacement>('customer');
  const [editingDetailId, setEditingDetailId] = useState<string | null>(null);

  const [newSku, setNewSku] = useState('');
  const [newProdName, setNewProdName] = useState('');
  const [newCategory, setNewCategory] = useState('Electronics');
  const [newPrice, setNewPrice] = useState(1500);
  const [newStock, setNewStock] = useState(50);
  const [productError, setProductError] = useState('');

  const [newReason, setNewReason] = useState('');

  const handleSaveGeneral = (e: React.FormEvent) => {
    e.preventDefault();
    onSaveSiteConfig(
      {
        ...draft,
        brandName: draft.brandName.trim() || 'ROUTEX',
        systemSubtitle: draft.systemSubtitle.trim() || 'Return Management System',
        loginTitle: draft.loginTitle.trim() || 'เข้าสู่ระบบ RouteX',
        supportLineId: draft.supportLineId.trim() || '140525455',
        supportPhone: draft.supportPhone.trim() || '0639569177',
      },
      'บันทึกการตั้งค่าข้อมูลเว็บไซต์และช่องทางติดต่อ'
    );
  };

  const handleSaveCustomDetail = (e: React.FormEvent) => {
    e.preventDefault();
    if (!detailTitle.trim() || !detailContent.trim()) return;

    let nextDetails: CustomDetailItem[];
    if (editingDetailId) {
      nextDetails = draft.customDetails.map((d) =>
        d.id === editingDetailId
          ? {
              ...d,
              title: detailTitle.trim(),
              content: detailContent.trim(),
              placement: detailPlacement,
            }
          : d
      );
    } else {
      nextDetails = [
        ...draft.customDetails,
        {
          id: `detail-${Date.now()}`,
          title: detailTitle.trim(),
          content: detailContent.trim(),
          placement: detailPlacement,
        },
      ];
    }

    const nextConfig: SiteConfig = { ...draft, customDetails: nextDetails };
    setDraft(nextConfig);
    onSaveSiteConfig(
      nextConfig,
      editingDetailId
        ? `แก้ไขรายละเอียดเพิ่มเติม "${detailTitle.trim()}"`
        : `เพิ่มรายละเอียดเพิ่มเติม "${detailTitle.trim()}"`
    );
    setDetailTitle('');
    setDetailContent('');
    setDetailPlacement('customer');
    setEditingDetailId(null);
  };

  const handleDeleteDetail = (item: CustomDetailItem) => {
    const nextConfig: SiteConfig = {
      ...draft,
      customDetails: draft.customDetails.filter((d) => d.id !== item.id),
    };
    setDraft(nextConfig);
    onSaveSiteConfig(nextConfig, `ลบหัวข้อรายละเอียด "${item.title}"`);
  };

  const handleCreateProduct = (e: React.FormEvent) => {
    e.preventDefault();
    if (!newSku.trim() || !newProdName.trim()) {
      setProductError('กรุณากรอกรหัส SKU และชื่อสินค้า');
      return;
    }

    const err = onAddProduct(
      {
        sku: newSku.trim().toUpperCase(),
        name: newProdName.trim(),
        category: newCategory.trim() || 'General',
        price: Math.max(0, Number(newPrice) || 0),
        stock: Math.max(0, Number(newStock) || 0),
        imageUrl: GENERATED_IMAGES.parcel,
      },
      () => {
        setNewSku('');
        setNewProdName('');
        setNewCategory('Electronics');
        setNewPrice(1500);
        setNewStock(50);
      }
    );

    if (err) {
      setProductError(err);
    } else {
      setProductError('');
    }
  };

  const handleAddReason = (e: React.FormEvent) => {
    e.preventDefault();
    const trimmed = newReason.trim();
    if (!trimmed || returnReasons.includes(trimmed)) return;
    onUpdateReturnReasons([...returnReasons, trimmed], `เพิ่มสาเหตุการคืน "${trimmed}"`);
    setNewReason('');
  };

  const handleRemoveReason = (reason: string) => {
    if (returnReasons.length <= 1) return;
    onUpdateReturnReasons(
      returnReasons.filter((r) => r !== reason),
      `ลบสาเหตุการคืน "${reason}"`
    );
  };

  const placementLabel: Record<CustomDetailPlacement, string> = {
    customer: 'แสดงในหน้าลูกค้า',
    all: 'แสดงทุกหน้าภายในระบบ',
    login: 'แสดงในหน้าแรกเข้าสู่ระบบ',
  };

  return (
    <div className="space-y-8 max-w-5xl">
      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
        <div>
          <h1 className="text-2xl font-bold text-slate-900">
            เปลี่ยนแปลงและเพิ่มรายละเอียดเว็บไซต์ (Admin 1)
          </h1>
          <p className="text-sm text-slate-500 mt-1">
            ปรับแก้ไขข้อมูลแบรนด์ ช่องทางติดต่อ เพิ่มหัวข้อประกาศ/รายละเอียด รวมถึงจัดการรายการสินค้าและสาเหตุการคืนได้อย่างอิสระ
          </p>
        </div>

        <button
          type="button"
          onClick={() => {
            onResetSiteConfig();
          }}
          className="inline-flex items-center gap-2 px-3.5 py-2 bg-white hover:bg-rose-50 text-rose-700 border border-rose-200 text-xs font-medium rounded-lg transition-colors whitespace-nowrap shrink-0 cursor-pointer"
        >
          <RotateCcw size={14} aria-hidden="true" />
          <span>คืนค่าข้อมูลเริ่มต้น</span>
        </button>
      </div>

      {/* 1. General Site Info & Customer Contact */}
      <form
        onSubmit={handleSaveGeneral}
        className="bg-white border border-rose-100 rounded-xl p-6 space-y-5"
      >
        <div className="flex items-center gap-2 border-b border-rose-100 pb-3">
          <Globe size={18} className="text-pink-600" aria-hidden="true" />
          <h2 className="text-base font-semibold text-slate-900">
            1. ข้อมูลระบบ ข้อความหน้าแรก และช่องทางติดต่อผู้ดูแล
          </h2>
        </div>

        <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
          <div className="space-y-1.5">
            <label htmlFor="cfg-brand" className="block text-xs font-medium text-slate-700">
              ชื่อแบรนด์ / โลโก้เว็บไซต์
            </label>
            <input
              id="cfg-brand"
              type="text"
              required
              value={draft.brandName}
              onChange={(e) => setDraft({ ...draft, brandName: e.target.value })}
              className="w-full px-3 py-2 text-sm border border-rose-200 rounded-lg focus:border-pink-500"
            />
          </div>

          <div className="space-y-1.5">
            <label htmlFor="cfg-sub" className="block text-xs font-medium text-slate-700">
              คำอธิบายระบบส่วนหัว (Header Subtitle)
            </label>
            <input
              id="cfg-sub"
              type="text"
              required
              value={draft.systemSubtitle}
              onChange={(e) => setDraft({ ...draft, systemSubtitle: e.target.value })}
              className="w-full px-3 py-2 text-sm border border-rose-200 rounded-lg focus:border-pink-500"
            />
          </div>

          <div className="space-y-1.5">
            <label htmlFor="cfg-login-title" className="block text-xs font-medium text-slate-700">
              หัวข้อหลักหน้าเข้าสู่ระบบ
            </label>
            <input
              id="cfg-login-title"
              type="text"
              required
              value={draft.loginTitle}
              onChange={(e) => setDraft({ ...draft, loginTitle: e.target.value })}
              className="w-full px-3 py-2 text-sm border border-rose-200 rounded-lg focus:border-pink-500"
            />
          </div>

          <div className="space-y-1.5">
            <label htmlFor="cfg-login-desc" className="block text-xs font-medium text-slate-700">
              คำอธิบายใต้หัวข้อหน้าเข้าสู่ระบบ
            </label>
            <input
              id="cfg-login-desc"
              type="text"
              value={draft.loginDescription}
              onChange={(e) => setDraft({ ...draft, loginDescription: e.target.value })}
              className="w-full px-3 py-2 text-sm border border-rose-200 rounded-lg focus:border-pink-500"
            />
          </div>

          <div className="space-y-1.5">
            <label htmlFor="cfg-line" className="block text-xs font-medium text-slate-700">
              LINE ID ผู้ดูแลระบบ (สำหรับลูกค้า)
            </label>
            <input
              id="cfg-line"
              type="text"
              required
              value={draft.supportLineId}
              onChange={(e) => setDraft({ ...draft, supportLineId: e.target.value })}
              className="w-full px-3 py-2 text-sm border border-rose-200 rounded-lg font-mono focus:border-pink-500"
            />
          </div>

          <div className="space-y-1.5">
            <label htmlFor="cfg-phone" className="block text-xs font-medium text-slate-700">
              เบอร์โทรศัพท์ติดต่อผู้ดูแลระบบ
            </label>
            <input
              id="cfg-phone"
              type="text"
              required
              value={draft.supportPhone}
              onChange={(e) => setDraft({ ...draft, supportPhone: e.target.value })}
              className="w-full px-3 py-2 text-sm border border-rose-200 rounded-lg font-mono focus:border-pink-500"
            />
          </div>

          <div className="sm:col-span-2 space-y-1.5">
            <label htmlFor="cfg-note" className="block text-xs font-medium text-slate-700">
              หมายเหตุหรือเวลาทำการติดต่อผู้ดูแล (แสดงในหน้าลูกค้า)
            </label>
            <input
              id="cfg-note"
              type="text"
              value={draft.supportNote}
              onChange={(e) => setDraft({ ...draft, supportNote: e.target.value })}
              placeholder="เช่น เปิดให้บริการทุกวัน 08:30 - 18:00 น."
              className="w-full px-3 py-2 text-sm border border-rose-200 rounded-lg focus:border-pink-500"
            />
          </div>
        </div>

        <div className="flex justify-end pt-2">
          <button
            type="submit"
            className="inline-flex items-center gap-2 px-5 py-2.5 bg-gradient-to-r from-pink-500 to-rose-500 hover:from-pink-600 hover:to-rose-600 text-white text-sm font-semibold rounded-lg shadow-sm shadow-pink-500/20 transition-all cursor-pointer"
          >
            <Save size={15} aria-hidden="true" />
            <span>บันทึกการเปลี่ยนแปลงข้อมูลเว็บไซต์</span>
          </button>
        </div>
      </form>

      {/* 2. Custom Details / Announcements */}
      <div className="bg-white border border-rose-100 rounded-xl p-6 space-y-6">
        <div className="flex items-center gap-2 border-b border-rose-100 pb-3">
          <FilePlus2 size={18} className="text-emerald-600" aria-hidden="true" />
          <h2 className="text-base font-semibold text-slate-900">
            2. เพิ่มหัวข้อรายละเอียด / ประกาศเพิ่มเติมบนเว็บไซต์
          </h2>
        </div>

        <form onSubmit={handleSaveCustomDetail} className="space-y-4">
          <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
            <div className="sm:col-span-2 space-y-1.5">
              <label htmlFor="det-title" className="block text-xs font-medium text-slate-700">
                หัวข้อรายละเอียด *
              </label>
              <input
                id="det-title"
                type="text"
                required
                value={detailTitle}
                onChange={(e) => setDetailTitle(e.target.value)}
                placeholder="เช่น เงื่อนไขการรับประกันและการแพ็กสินค้าคืน"
                className="w-full px-3 py-2 text-sm border border-rose-200 rounded-lg focus:border-pink-500"
              />
            </div>

            <div className="space-y-1.5">
              <label htmlFor="det-place" className="block text-xs font-medium text-slate-700">
                ตำแหน่งที่แสดงผล
              </label>
              <select
                id="det-place"
                value={detailPlacement}
                onChange={(e) => setDetailPlacement(e.target.value as CustomDetailPlacement)}
                className="w-full px-3 py-2 text-sm border border-rose-200 rounded-lg bg-white focus:border-pink-500"
              >
                <option value="customer">แสดงในหน้าลูกค้า</option>
                <option value="all">แสดงทุกหน้าภายในระบบ</option>
                <option value="login">แสดงในหน้าแรกเข้าสู่ระบบ</option>
              </select>
            </div>
          </div>

          <div className="space-y-1.5">
            <label htmlFor="det-content" className="block text-xs font-medium text-slate-700">
              เนื้อหารายละเอียด *
            </label>
            <textarea
              id="det-content"
              rows={3}
              required
              value={detailContent}
              onChange={(e) => setDetailContent(e.target.value)}
              placeholder="ระบุข้อความรายละเอียด นโยบาย หรือประกาศที่ต้องการแสดงให้ผู้ใช้งานเห็น..."
              className="w-full px-3 py-2 text-sm border border-rose-200 rounded-lg focus:border-pink-500"
            />
          </div>

          <div className="flex items-center justify-end gap-2">
            {editingDetailId && (
              <button
                type="button"
                onClick={() => {
                  setEditingDetailId(null);
                  setDetailTitle('');
                  setDetailContent('');
                  setDetailPlacement('customer');
                }}
                className="px-4 py-2 bg-rose-50 hover:bg-rose-100 text-rose-700 text-xs font-medium rounded-lg cursor-pointer"
              >
                ยกเลิกการแก้ไข
              </button>
            )}
            <button
              type="submit"
              className="inline-flex items-center gap-2 px-4 py-2 bg-emerald-600 hover:bg-emerald-700 text-white text-sm font-semibold rounded-lg transition-colors cursor-pointer"
            >
              <Plus size={15} aria-hidden="true" />
              <span>{editingDetailId ? 'บันทึกการแก้ไขหัวข้อ' : 'เพิ่มรายละเอียดลงเว็บไซต์'}</span>
            </button>
          </div>
        </form>

        <div className="border-t border-rose-100 pt-4 space-y-3">
          <p className="text-xs font-semibold text-slate-700">
            รายการหัวข้อที่เพิ่มไว้ ({draft.customDetails.length} รายการ)
          </p>
          {draft.customDetails.length === 0 ? (
            <p className="text-xs text-slate-500 py-3">
              ยังไม่มีหัวข้อรายละเอียดเพิ่มเติมที่สร้างไว้
            </p>
          ) : (
            <div className="divide-y divide-rose-100 border border-rose-200 rounded-lg">
              {draft.customDetails.map((item) => (
                <div
                  key={item.id}
                  className="p-4 flex flex-col sm:flex-row sm:items-start justify-between gap-3"
                >
                  <div className="space-y-1">
                    <div className="flex flex-wrap items-center gap-2">
                      <span className="text-sm font-semibold text-slate-900">
                        {item.title}
                      </span>
                      <span className="text-xs text-pink-700 font-medium">
                        · {placementLabel[item.placement]}
                      </span>
                    </div>
                    <p className="text-xs text-slate-600 leading-relaxed whitespace-pre-line">
                      {item.content}
                    </p>
                  </div>
                  <div className="flex items-center gap-1.5 shrink-0">
                    <button
                      type="button"
                      onClick={() => {
                        setEditingDetailId(item.id);
                        setDetailTitle(item.title);
                        setDetailContent(item.content);
                        setDetailPlacement(item.placement);
                      }}
                      className="inline-flex items-center gap-1 px-2.5 py-1.5 bg-rose-50 hover:bg-rose-100 text-rose-700 text-xs font-medium rounded-md cursor-pointer"
                    >
                      <Edit3 size={13} aria-hidden="true" />
                      <span>แก้ไข</span>
                    </button>
                    <button
                      type="button"
                      onClick={() => handleDeleteDetail(item)}
                      className="inline-flex items-center gap-1 px-2.5 py-1.5 text-rose-700 hover:bg-rose-50 text-xs font-medium rounded-md cursor-pointer"
                    >
                      <Trash2 size={13} aria-hidden="true" />
                      <span>ลบ</span>
                    </button>
                  </div>
                </div>
              ))}
            </div>
          )}
        </div>
      </div>

      {/* 3. Manage SKU Catalog & Return Reasons */}
      <div className="grid grid-cols-1 lg:grid-cols-2 gap-6 items-start">
        <div className="bg-white border border-rose-100 rounded-xl p-6 space-y-5">
          <div className="flex items-center gap-2 border-b border-rose-100 pb-3">
            <PackagePlus size={18} className="text-pink-600" aria-hidden="true" />
            <h2 className="text-base font-semibold text-slate-900">
              3. จัดการรายการสินค้า (SKU Catalog)
            </h2>
          </div>

          {productError && (
            <div className="p-3 rounded-lg bg-rose-50 border border-rose-200 text-rose-800 text-xs">
              {productError}
            </div>
          )}

          <form onSubmit={handleCreateProduct} className="space-y-3">
            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
              <div className="space-y-1">
                <label htmlFor="prod-sku" className="block text-xs font-medium text-slate-700">
                  รหัส SKU *
                </label>
                <input
                  id="prod-sku"
                  type="text"
                  required
                  value={newSku}
                  onChange={(e) => setNewSku(e.target.value)}
                  placeholder="SKU-NEW-001"
                  className="w-full px-3 py-2 text-xs border border-rose-200 rounded-lg font-mono focus:border-pink-500"
                />
              </div>
              <div className="space-y-1">
                <label htmlFor="prod-cat" className="block text-xs font-medium text-slate-700">
                  หมวดหมู่
                </label>
                <input
                  id="prod-cat"
                  type="text"
                  value={newCategory}
                  onChange={(e) => setNewCategory(e.target.value)}
                  className="w-full px-3 py-2 text-xs border border-rose-200 rounded-lg focus:border-pink-500"
                />
              </div>
              <div className="sm:col-span-2 space-y-1">
                <label htmlFor="prod-name" className="block text-xs font-medium text-slate-700">
                  ชื่อสินค้า *
                </label>
                <input
                  id="prod-name"
                  type="text"
                  required
                  value={newProdName}
                  onChange={(e) => setNewProdName(e.target.value)}
                  placeholder="ชื่อสินค้าที่ต้องการเพิ่มเข้าระบบ"
                  className="w-full px-3 py-2 text-xs border border-rose-200 rounded-lg focus:border-pink-500"
                />
              </div>
              <div className="space-y-1">
                <label htmlFor="prod-price" className="block text-xs font-medium text-slate-700">
                  ราคา (บาท)
                </label>
                <input
                  id="prod-price"
                  type="number"
                  min={0}
                  value={newPrice}
                  onChange={(e) => setNewPrice(Number(e.target.value) || 0)}
                  className="w-full px-3 py-2 text-xs border border-rose-200 rounded-lg font-mono tabular-nums focus:border-pink-500"
                />
              </div>
              <div className="space-y-1">
                <label htmlFor="prod-stock" className="block text-xs font-medium text-slate-700">
                  จำนวนสต็อกตั้งต้น
                </label>
                <input
                  id="prod-stock"
                  type="number"
                  min={0}
                  value={newStock}
                  onChange={(e) => setNewStock(Number(e.target.value) || 0)}
                  className="w-full px-3 py-2 text-xs border border-rose-200 rounded-lg font-mono tabular-nums focus:border-pink-500"
                />
              </div>
            </div>

            <button
              type="submit"
              className="w-full py-2 bg-pink-600 hover:bg-pink-700 text-white text-xs font-semibold rounded-lg transition-colors cursor-pointer"
            >
              เพิ่มสินค้าเข้าระบบ
            </button>
          </form>

          <div className="divide-y divide-rose-100 border border-rose-200 rounded-lg max-h-64 overflow-y-auto">
            {products.map((p) => (
              <div key={p.sku} className="px-3.5 py-2.5 flex items-center justify-between gap-2 text-xs">
                <div className="min-w-0">
                  <p className="font-medium text-slate-900 truncate">{p.name}</p>
                  <p className="font-mono text-slate-500">
                    {p.sku} · {formatCurrency(p.price)} · คงเหลือ {p.stock}
                  </p>
                </div>
                {products.length > 1 && (
                  <button
                    type="button"
                    onClick={() => onDeleteProduct(p.sku)}
                    aria-label={`ลบสินค้า ${p.sku}`}
                    className="p-1.5 text-rose-600 hover:bg-rose-50 rounded cursor-pointer shrink-0"
                  >
                    <Trash2 size={14} />
                  </button>
                )}
              </div>
            ))}
          </div>
        </div>

        <div className="bg-white border border-rose-100 rounded-xl p-6 space-y-5">
          <div className="flex items-center gap-2 border-b border-rose-100 pb-3">
            <Edit3 size={18} className="text-amber-600" aria-hidden="true" />
            <h2 className="text-base font-semibold text-slate-900">
              4. ตัวเลือกสาเหตุการคืนสินค้า (ฟอร์มลูกค้า)
            </h2>
          </div>

          <form onSubmit={handleAddReason} className="flex gap-2">
            <input
              type="text"
              value={newReason}
              onChange={(e) => setNewReason(e.target.value)}
              placeholder="เพิ่มตัวเลือกสาเหตุการคืนใหม่..."
              className="flex-1 px-3 py-2 text-xs border border-rose-200 rounded-lg focus:border-pink-500"
            />
            <button
              type="submit"
              className="px-3.5 py-2 bg-pink-600 hover:bg-pink-700 text-white text-xs font-semibold rounded-lg whitespace-nowrap cursor-pointer"
            >
              เพิ่มตัวเลือก
            </button>
          </form>

          <div className="divide-y divide-rose-100 border border-rose-200 rounded-lg">
            {returnReasons.map((reason) => (
              <div
                key={reason}
                className="px-3.5 py-2.5 flex items-center justify-between gap-2 text-xs"
              >
                <span className="text-slate-800 font-medium">{reason}</span>
                {returnReasons.length > 1 && (
                  <button
                    type="button"
                    onClick={() => handleRemoveReason(reason)}
                    aria-label={`ลบสาเหตุ ${reason}`}
                    className="p-1 text-rose-600 hover:bg-rose-50 rounded cursor-pointer"
                  >
                    <Trash2 size={13} />
                  </button>
                )}
              </div>
            ))}
          </div>
        </div>
      </div>
    </div>
  );
}

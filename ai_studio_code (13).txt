import React, { useEffect, useState } from 'react';
import {
  ArrowRight,
  Banknote,
  CheckCircle2,
  Clock,
  Mail,
  MapPin,
  PackageCheck,
  Search,
  Send,
  ShieldCheck,
  Truck,
  Upload,
  XCircle,
} from 'lucide-react';
import {
  formatCurrency,
  ReturnRequest,
  ReturnStatus,
  Role,
  ROLE_THAI_LABELS,
  SiteConfig,
  STATUS_WORKFLOW,
  STEP_APPROVER_ROLE,
} from '../types';
import { ResilientImage, StatusIndicator } from './StatusText';

export interface SendRefundSlipPayload {
  sentToEmail: string;
  amount: number;
  transferRef: string;
  slipImageUrl?: string;
  note?: string;
  closeAfterSend: boolean;
}

interface TrackingViewProps {
  requests: ReturnRequest[];
  selectedRequest: ReturnRequest | null;
  onSelectRequest: (req: ReturnRequest | null) => void;
  activeRole: Role;
  siteConfig: SiteConfig;
  onUpdateStatus: (id: string, nextStatus: ReturnStatus) => void;
  onReceiveWarehouse?: (requestId: string, warehouseId: string, locationBin: string) => void;
  onSaveCustomerReturnTracking: (
    requestId: string,
    trackingNumber: string,
    carrier: string
  ) => void;
  onSaveCustomerEmail?: (requestId: string, email: string) => void;
  onSendRefundSlip?: (
    requestId: string,
    payload: SendRefundSlipPayload,
    onSuccess: () => void
  ) => void;
}

const CUSTOMER_CARRIERS = [
  'Flash Express',
  'Kerry Express',
  'Thailand Post (EMS)',
  'J&T Express',
  'SPX Express',
  'ขนส่งอื่นๆ',
];

export function TrackingView({
  requests,
  selectedRequest,
  onSelectRequest,
  activeRole,
  siteConfig,
  onUpdateStatus,
  onReceiveWarehouse,
  onSaveCustomerReturnTracking,
  onSaveCustomerEmail,
  onSendRefundSlip,
}: TrackingViewProps) {
  const isCustomer = activeRole === 'Customer';
  const activeReq = isCustomer ? selectedRequest : selectedRequest || requests[0] || null;

  const [trackingQuery, setTrackingQuery] = useState<string>(
    selectedRequest?.customerReturnTracking || selectedRequest?.id || ''
  );
  const [searchError, setSearchError] = useState<string>('');

  const [returnParcelInput, setReturnParcelInput] = useState<string>(
    activeReq?.customerReturnTracking || ''
  );
  const [returnCarrierInput, setReturnCarrierInput] = useState<string>(
    activeReq?.customerReturnCarrier || CUSTOMER_CARRIERS[0]
  );

  const [customerEmailInput, setCustomerEmailInput] = useState<string>(
    activeReq?.customerEmail || ''
  );
  const [slipEmail, setSlipEmail] = useState<string>(activeReq?.customerEmail || '');
  const [slipAmount, setSlipAmount] = useState<number>(
    activeReq ? activeReq.unitPrice * activeReq.qty : 0
  );
  const [slipRef, setSlipRef] = useState<string>('');
  const [slipImageUrl, setSlipImageUrl] = useState<string>('');
  const [slipNote, setSlipNote] = useState<string>('');
  const [slipCloseAfter, setSlipCloseAfter] = useState<boolean>(true);
  const [slipError, setSlipError] = useState<string>('');

  useEffect(() => {
    setReturnParcelInput(activeReq?.customerReturnTracking || '');
    setReturnCarrierInput(activeReq?.customerReturnCarrier || CUSTOMER_CARRIERS[0]);
    setCustomerEmailInput(activeReq?.customerEmail || '');
    setSlipEmail(activeReq?.customerEmail || '');
    setSlipAmount(activeReq ? activeReq.unitPrice * activeReq.qty : 0);
    setSlipRef('');
    setSlipImageUrl('');
    setSlipNote('');
    setSlipError('');
    if (isCustomer && selectedRequest) {
      setTrackingQuery(
        selectedRequest.customerReturnTracking || selectedRequest.id
      );
      setSearchError('');
    }
  }, [activeReq?.id, activeReq?.customerReturnTracking, activeReq?.customerReturnCarrier, activeReq?.customerEmail, isCustomer, selectedRequest]);

  const handleTrackingSearch = (e: React.FormEvent) => {
    e.preventDefault();
    const q = trackingQuery.trim().toLowerCase();
    if (!q) {
      setSearchError('กรุณากรอกเลขพัสดุ รหัสติดตาม (RTX) หรือเลขคำสั่งซื้อ');
      return;
    }

    const matched = requests.find(
      (r) =>
        r.id.toLowerCase() === q ||
        r.orderNumber.toLowerCase() === q ||
        (r.customerReturnTracking && r.customerReturnTracking.toLowerCase() === q) ||
        (r.outboundDispatch?.trackingNumber &&
          r.outboundDispatch.trackingNumber.toLowerCase() === q)
    );

    if (!matched) {
      onSelectRequest(null);
      setSearchError(
        `ไม่พบข้อมูลพัสดุหมายเลข "${trackingQuery.trim()}" กรุณาตรวจสอบเลขพัสดุอีกครั้ง`
      );
      return;
    }

    setSearchError('');
    onSelectRequest(matched);
  };

  const handleSaveParcel = (e: React.FormEvent) => {
    e.preventDefault();
    if (!activeReq || !returnParcelInput.trim()) return;
    onSaveCustomerReturnTracking(
      activeReq.id,
      returnParcelInput.trim(),
      returnCarrierInput
    );
  };

  const handleSaveEmail = (e: React.FormEvent) => {
    e.preventDefault();
    if (!activeReq || !customerEmailInput.trim() || !onSaveCustomerEmail) return;
    onSaveCustomerEmail(activeReq.id, customerEmailInput.trim());
  };

  const handleSlipFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = () => {
      if (typeof reader.result === 'string') {
        setSlipImageUrl(reader.result);
      }
    };
    reader.readAsDataURL(file);
  };

  const handleSendSlipSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (!activeReq || !onSendRefundSlip) return;
    if (!slipEmail.trim() || !slipRef.trim() || slipAmount <= 0 || !slipImageUrl) {
      setSlipError('กรุณากรอกอีเมล ยอดเงินคืน เลขอ้างอิงสลิป และแนบรูปภาพสลิปโอนเงินให้ครบถ้วน');
      return;
    }
    setSlipError('');
    onSendRefundSlip(
      activeReq.id,
      {
        sentToEmail: slipEmail.trim(),
        amount: slipAmount,
        transferRef: slipRef.trim(),
        slipImageUrl: slipImageUrl || undefined,
        note: slipNote.trim() || undefined,
        closeAfterSend: slipCloseAfter,
      },
      () => {
        setSlipRef('');
        setSlipImageUrl('');
        setSlipNote('');
      }
    );
  };

  if (!activeReq) {
    return (
      <div className="space-y-6">
        <div>
          <h1 className="text-2xl font-bold text-slate-900">
            ติดตามสถานะพัสดุคืน (7-Steps Workflow)
          </h1>
          <p className="text-sm text-slate-500 mt-1">
            กรอกเลขพัสดุของคุณเพื่อตรวจสอบสถานะการคืนสินค้าแบบเรียลไทม์
          </p>
        </div>

        <div className="bg-white border border-rose-100 rounded-xl p-6 sm:p-10 max-w-2xl mx-auto space-y-5 shadow-xs shadow-pink-500/5">
          <div className="text-center space-y-2">
            <div className="w-12 h-12 rounded-xl bg-rose-50 border border-rose-200 text-pink-600 flex items-center justify-center mx-auto">
              <Search size={22} aria-hidden="true" />
            </div>
            <h2 className="text-lg font-bold text-slate-900">
              ค้นหาและติดตามสถานะด้วยเลขพัสดุของคุณ
            </h2>
            <p className="text-xs sm:text-sm text-slate-500">
              กรอกเลขพัสดุที่ลูกค้าจัดส่งคืน, รหัสติดตามคำร้อง (RTX-...) หรือเลขคำสั่งซื้อ (ORD-...)
            </p>
          </div>

          <form onSubmit={handleTrackingSearch} className="space-y-3">
            <div className="flex flex-col sm:flex-row gap-2.5">
              <div className="relative flex-1">
                <Search
                  size={16}
                  className="absolute left-3.5 top-1/2 -translate-y-1/2 text-rose-400"
                  aria-hidden="true"
                />
                <input
                  type="text"
                  value={trackingQuery}
                  onChange={(e) => {
                    setTrackingQuery(e.target.value);
                    setSearchError('');
                  }}
                  placeholder="กรอกเลขพัสดุของลูกค้า เช่น TH0123456789 หรือ RTX-..."
                  aria-label="กรอกเลขพัสดุของลูกค้า"
                  className="w-full pl-10 pr-3 py-2.5 text-sm border border-rose-200 rounded-lg bg-rose-50/30 focus:bg-white font-mono text-slate-900 focus:border-pink-500"
                />
              </div>
              <button
                type="submit"
                className="px-5 py-2.5 bg-gradient-to-r from-pink-500 to-rose-500 hover:from-pink-600 hover:to-rose-600 text-white text-sm font-semibold rounded-lg shadow-sm shadow-pink-500/20 transition-all whitespace-nowrap cursor-pointer"
              >
                ตรวจสอบสถานะ
              </button>
            </div>

            {searchError && (
              <div
                role="alert"
                className="p-3 rounded-lg bg-rose-50 border border-rose-200 text-rose-800 text-xs"
              >
                {searchError}
              </div>
            )}
          </form>

          {requests.length === 0 ? (
            <p className="text-center text-xs text-slate-500">
              ยังไม่มีรายการคำขอคืนสินค้าในระบบ
            </p>
          ) : (
            <div className="pt-4 border-t border-rose-100 text-xs text-slate-500 text-center space-y-1">
              <p>
                หากยังไม่มีเลขติดตาม สามารถยื่นคำขอคืนสินค้าได้ที่เมนู{' '}
                <span className="font-semibold text-pink-700">"ยื่นคำขอคืนสินค้า"</span>{' '}
                เพื่อรับรหัสติดตาม
              </p>
              <p>
                ติดต่อสอบถาม: LINE{' '}
                <span className="font-mono font-semibold text-pink-700">
                  {siteConfig.supportLineId}
                </span>{' '}
                · โทร{' '}
                <a
                  href={`tel:${siteConfig.supportPhone}`}
                  className="font-mono font-semibold text-pink-700 hover:underline"
                >
                  {siteConfig.supportPhone}
                </a>
              </p>
            </div>
          )}
        </div>
      </div>
    );
  }

  const currentStepIndex = STATUS_WORKFLOW.indexOf(activeReq.status);
  const isRejected = activeReq.status === 'ปฏิเสธคำขอคืนสินค้า';
  const nextWorkflowStep =
    currentStepIndex >= 0 && currentStepIndex < STATUS_WORKFLOW.length - 1
      ? STATUS_WORKFLOW[currentStepIndex + 1]
      : null;

  return (
    <div className="space-y-6">
      <div className="flex flex-col md:flex-row md:items-center md:justify-between gap-4">
        <div>
          <h1 className="text-2xl font-bold text-slate-900">
            ติดตามสถานะพัสดุคืน (7-Steps Workflow)
          </h1>
          <p className="text-sm text-slate-500 mt-1">
            ตรวจสอบความคืบหน้าตั้งแต่ยื่นคำร้อง รับพัสดุเข้าคลัง ตรวจสภาพ จนถึงการปิดรายการ
          </p>
        </div>

        {isCustomer ? (
          <form onSubmit={handleTrackingSearch} className="flex items-center gap-2 w-full md:w-auto">
            <div className="relative flex-1 md:w-72">
              <Search
                size={15}
                className="absolute left-3 top-1/2 -translate-y-1/2 text-rose-400"
                aria-hidden="true"
              />
              <input
                type="text"
                value={trackingQuery}
                onChange={(e) => {
                  setTrackingQuery(e.target.value);
                  setSearchError('');
                }}
                placeholder="กรอกเลขพัสดุ / รหัสติดตามของคุณ"
                aria-label="กรอกเลขพัสดุของลูกค้า"
                className="w-full pl-9 pr-3 py-2 text-sm border border-rose-200 rounded-lg bg-white font-mono text-slate-800 focus:border-pink-500"
              />
            </div>
            <button
              type="submit"
              className="px-4 py-2 bg-pink-600 hover:bg-pink-700 text-white text-sm font-medium rounded-lg whitespace-nowrap shrink-0 cursor-pointer"
            >
              ค้นหา
            </button>
          </form>
        ) : (
          <div className="flex items-center gap-2">
            <Search size={16} className="text-rose-400 shrink-0" aria-hidden="true" />
            <select
              aria-label="เลือกรายการคำขอคืนสินค้าเพื่อติดตามสถานะ"
              value={activeReq.id}
              onChange={(e) => {
                const found = requests.find((r) => r.id === e.target.value);
                if (found) onSelectRequest(found);
              }}
              className="px-3 py-2 text-sm border border-rose-200 rounded-lg bg-white font-mono text-slate-800 focus:border-pink-500"
            >
              {requests.map((r) => (
                <option key={r.id} value={r.id}>
                  {r.id} · {r.customerName}
                </option>
              ))}
            </select>
          </div>
        )}
      </div>

      {isCustomer && searchError && (
        <div
          role="alert"
          className="p-3.5 rounded-lg bg-rose-50 border border-rose-200 text-rose-800 text-xs"
        >
          {searchError}
        </div>
      )}

      <div className="bg-white border border-rose-100 rounded-xl p-6 space-y-6">
        <div className="flex flex-col lg:flex-row lg:items-center lg:justify-between gap-4 border-b border-rose-100 pb-5">
          <div className="space-y-1">
            <div className="flex flex-wrap items-center gap-3">
              <h2 className="text-xl font-bold font-mono text-slate-900">{activeReq.id}</h2>
              <StatusIndicator status={activeReq.status} />
            </div>
            <p className="text-xs text-slate-500 font-mono">
              คำสั่งซื้อ {activeReq.orderNumber} · ยื่นคำร้องวันที่ {activeReq.requestDate} · ผู้ยื่นคำร้อง {activeReq.customerName}
            </p>
          </div>

          {(() => {
            if (!nextWorkflowStep || isRejected) return null;
            const nextApprover = STEP_APPROVER_ROLE[nextWorkflowStep];
            const needsRefundSlip =
              nextWorkflowStep === 'ตรวจสอบเสร็จสิ้น / ปิดรายการ' &&
              activeReq.resolution?.type === 'refund' &&
              (!activeReq.refundSlip?.transferRef?.trim() ||
                !activeReq.refundSlip?.slipImageUrl);

            if (activeRole === nextApprover) {
              if (needsRefundSlip) {
                return (
                  <span className="px-3.5 py-2 rounded-lg bg-amber-50 border border-amber-200 text-xs font-medium text-amber-800 whitespace-nowrap shrink-0">
                    กรุณาส่งสลิปและกรอกเลขอ้างอิงด้านล่างก่อน จึงจะปิดรายการได้
                  </span>
                );
              }
              return (
                <button
                  type="button"
                  onClick={() => {
                    if (
                      nextWorkflowStep === 'ได้รับสินค้าแล้ว' &&
                      onReceiveWarehouse
                    ) {
                      onReceiveWarehouse(
                        activeReq.id,
                        activeReq.warehouseId || 'WH-BKK-01',
                        activeReq.locationBin === 'PENDING-BIN'
                          ? 'RACK-A01'
                          : activeReq.locationBin
                      );
                    } else {
                      onUpdateStatus(activeReq.id, nextWorkflowStep);
                    }
                  }}
                  className="inline-flex items-center gap-2 px-4 py-2 bg-pink-600 hover:bg-pink-700 text-white text-sm font-medium rounded-lg shadow-xs shadow-pink-500/15 transition-colors whitespace-nowrap shrink-0 cursor-pointer"
                >
                  <span>อนุมัติขั้นตอนถัดไป: {nextWorkflowStep}</span>
                  <ArrowRight size={15} aria-hidden="true" />
                </button>
              );
            }
            if (activeRole !== 'Customer') {
              return (
                <span className="px-3.5 py-2 rounded-lg bg-rose-50 border border-rose-200 text-xs font-medium text-rose-800 whitespace-nowrap shrink-0">
                  รอ{ROLE_THAI_LABELS[nextApprover]}อนุมัติ: {nextWorkflowStep}
                </span>
              );
            }
            return null;
          })()}
        </div>

        {isRejected ? (
          <div className="p-5 rounded-xl bg-rose-50 border border-rose-200 flex items-start gap-3">
            <XCircle size={22} className="text-rose-600 shrink-0 mt-0.5" aria-hidden="true" />
            <div>
              <h3 className="text-sm font-semibold text-rose-900">คำขอคืนสินค้าถูกปฏิเสธ</h3>
              <p className="text-xs text-rose-700 mt-1">
                คำร้องนี้ไม่ผ่านเงื่อนไขการรับคืนสินค้า กรุณาติดต่อผู้ดูแลระบบทาง LINE ID:{' '}
                <span className="font-mono font-semibold">{siteConfig.supportLineId}</span>{' '}
                หรือโทร{' '}
                <a
                  href={`tel:${siteConfig.supportPhone}`}
                  className="font-mono font-semibold underline"
                >
                  {siteConfig.supportPhone}
                </a>{' '}
                เพื่อสอบถามรายละเอียดเพิ่มเติม
              </p>
            </div>
          </div>
        ) : (
          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-7 gap-3">
            {STATUS_WORKFLOW.map((step, index) => {
              const isCompleted = index < currentStepIndex;
              const isCurrent = index === currentStepIndex;
              const stepApprover = ROLE_THAI_LABELS[STEP_APPROVER_ROLE[step]];
              return (
                <div
                  key={step}
                  className={`p-3.5 rounded-lg border transition-colors ${
                    isCurrent
                      ? 'bg-pink-50 border-pink-300'
                      : isCompleted
                        ? 'bg-emerald-50/70 border-emerald-200'
                        : 'bg-rose-50/40 border-rose-100'
                  }`}
                >
                  <div className="flex items-center justify-between mb-2">
                    <span
                      className={`text-xs font-mono font-semibold tabular-nums ${
                        isCurrent
                          ? 'text-pink-700'
                          : isCompleted
                            ? 'text-emerald-700'
                            : 'text-slate-400'
                      }`}
                    >
                      0{index + 1}
                    </span>
                    {isCompleted ? (
                      <CheckCircle2 size={15} className="text-emerald-600" aria-hidden="true" />
                    ) : isCurrent ? (
                      <Clock size={15} className="text-pink-600" aria-hidden="true" />
                    ) : (
                      <div className="w-3.5 h-3.5 rounded-full border border-rose-200" />
                    )}
                  </div>
                  <p
                    className={`text-xs font-medium leading-snug ${
                      isCurrent
                        ? 'text-pink-950 font-semibold'
                        : isCompleted
                          ? 'text-emerald-950'
                          : 'text-slate-500'
                    }`}
                  >
                    {step}
                  </p>
                  <p className="text-[11px] text-slate-400 mt-1.5">
                    ผู้อนุมัติ: {stepApprover}
                  </p>
                </div>
              );
            })}
          </div>
        )}
      </div>

      {!isRejected && activeReq.resolution?.type === 'refund' && (
        <div className="bg-white border border-emerald-200 rounded-xl p-6 space-y-4">
          <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2 border-b border-emerald-100 pb-3">
            <div className="flex items-center gap-2">
              <Mail size={17} className="text-emerald-600" aria-hidden="true" />
              <h3 className="text-sm font-semibold text-slate-900">
                ส่งสลิปโอนเงินคืนถึงอีเมลลูกค้า (ไม่ต้องกรอกแบบฟอร์มส่งกลับสินค้า)
              </h3>
            </div>
            {activeReq.refundSlip ? (
              <span className="inline-flex items-center gap-1.5 text-xs font-semibold text-emerald-700 bg-emerald-50 border border-emerald-200 px-2.5 py-1 rounded-md">
                <CheckCircle2 size={13} aria-hidden="true" />
                <span>ส่งสลิปถึงเมลลูกค้าแล้ว</span>
              </span>
            ) : (
              <span className="inline-flex items-center gap-1.5 text-xs font-medium text-amber-800 bg-amber-50 border border-amber-200 px-2.5 py-1 rounded-md">
                <Clock size={13} aria-hidden="true" />
                <span>รอส่งสลิปถึงอีเมลลูกค้า</span>
              </span>
            )}
          </div>

          {activeReq.refundSlip ? (
            <div className="bg-emerald-50/50 border border-emerald-200 rounded-xl p-4 grid grid-cols-1 md:grid-cols-3 gap-4 items-start">
              <div className="md:col-span-2 space-y-2 text-xs">
                <div className="flex items-center gap-2 text-sm font-semibold text-emerald-950">
                  <Banknote size={16} className="text-emerald-600" aria-hidden="true" />
                  <span>หลักฐานการโอนเงินคืนถูกส่งไปยังอีเมลลูกค้าแล้ว</span>
                </div>
                <dl className="grid grid-cols-1 sm:grid-cols-2 gap-2 pt-1">
                  <div>
                    <dt className="text-slate-500">ส่งสลิปไปยังอีเมล</dt>
                    <dd className="font-mono font-semibold text-slate-900">
                      {activeReq.refundSlip.sentToEmail}
                    </dd>
                  </div>
                  <div>
                    <dt className="text-slate-500">ยอดเงินที่โอนคืน</dt>
                    <dd className="font-mono font-bold text-emerald-700 text-sm">
                      {formatCurrency(activeReq.refundSlip.amount)}
                    </dd>
                  </div>
                  <div>
                    <dt className="text-slate-500">เลขอ้างอิงสลิป (Transaction Ref)</dt>
                    <dd className="font-mono font-semibold text-slate-900">
                      {activeReq.refundSlip.transferRef}
                    </dd>
                  </div>
                  <div>
                    <dt className="text-slate-500">วันเวลาที่ส่งสลิป</dt>
                    <dd className="font-mono text-slate-700">
                      {activeReq.refundSlip.sentAt} โดย {activeReq.refundSlip.sentBy}
                    </dd>
                  </div>
                </dl>
                {activeReq.resolution && (
                  <p className="text-slate-600 pt-1">
                    โอนเข้าบัญชี: {activeReq.resolution.refundBankName} ·{' '}
                    {activeReq.resolution.refundAccountName} (
                    <span className="font-mono font-semibold text-slate-900">

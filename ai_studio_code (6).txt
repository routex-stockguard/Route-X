import React, { useState } from 'react';
import {
  CheckCircle2,
  Clock,
  Package,
  ShieldCheck,
  Truck,
  XCircle,
  RefreshCw,
} from 'lucide-react';
import { ReturnStatus } from '../types';

interface StatusIndicatorProps {
  status: ReturnStatus;
}

export function StatusIndicator({ status }: StatusIndicatorProps) {
  let colorClasses = 'bg-slate-100 text-slate-700 border-slate-200';
  let Icon = Clock;

  switch (status) {
    case 'รออนุมัติคำขอคืนสินค้า':
      colorClasses = 'bg-amber-50 text-amber-800 border-amber-200';
      Icon = Clock;
      break;
    case 'อนุมัติการคืนสินค้า':
    case 'รอรับสินค้าคืน':
      colorClasses = 'bg-pink-50 text-pink-800 border-pink-200';
      Icon = Truck;
      break;
    case 'ได้รับสินค้าแล้ว':
    case 'อยู่ระหว่างตรวจสอบสินค้า':
      colorClasses = 'bg-fuchsia-50 text-fuchsia-800 border-fuchsia-200';
      Icon = ShieldCheck;
      break;
    case 'คืนเงิน / เปลี่ยนสินค้า':
      colorClasses = 'bg-rose-50 text-rose-700 border-rose-200';
      Icon = RefreshCw;
      break;
    case 'ตรวจสอบเสร็จสิ้น / ปิดรายการ':
      colorClasses = 'bg-emerald-50 text-emerald-800 border-emerald-200';
      Icon = CheckCircle2;
      break;
    case 'ปฏิเสธคำขอคืนสินค้า':
      colorClasses = 'bg-red-50 text-red-800 border-red-200';
      Icon = XCircle;
      break;
  }

  return (
    <span
      className={`inline-flex items-center gap-1.5 px-2.5 py-1 rounded-md text-xs font-medium border whitespace-nowrap ${colorClasses}`}
    >
      <Icon size={13} className="shrink-0" aria-hidden="true" />
      <span>{status}</span>
    </span>
  );
}

interface ResilientImageProps {
  src: string;
  alt: string;
  className?: string;
}

export function ResilientImage({ src, alt, className = 'w-full h-full object-cover' }: ResilientImageProps) {
  const [hasError, setHasError] = useState(false);

  if (!src || hasError) {
    return (
      <div className="w-full h-full bg-gradient-to-br from-rose-50 to-pink-100 flex flex-col items-center justify-center p-3 text-rose-700 text-center">
        <Package size={24} className="mb-1.5 text-pink-400" aria-hidden="true" />
        <span className="text-xs font-medium line-clamp-2">{alt}</span>
      </div>
    );
  }

  return (
    <img
      src={src}
      alt={alt}
      referrerPolicy="no-referrer"
      onError={() => setHasError(true)}
      className={className}
    />
  );
}

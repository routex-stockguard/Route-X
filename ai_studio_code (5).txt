export function formatCurrency(amount: number): string {
  return `฿${amount.toLocaleString('th-TH', { minimumFractionDigits: 0, maximumFractionDigits: 0 })}`;
}

export function getTotalRequestCost(req: ReturnRequest): number {
  return req.costs.freight + req.costs.repair + req.costs.inspectionFee + req.costs.assetLoss;
}

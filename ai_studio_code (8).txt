import React, { useState } from 'react';
import { ShieldCheck, X } from 'lucide-react';
import { CurrentUser, StaffUser } from '../types';

interface StaffVerifyModalProps {
  actionLabel: string;
  currentUser: CurrentUser;
  users: StaffUser[];
  onCancel: () => void;
  onVerified: (verifiedUser: StaffUser) => void;
}

export function StaffVerifyModal({
  actionLabel,
  currentUser,
  users,
  onCancel,
  onVerified,
}: StaffVerifyModalProps) {
  const [employeeId, setEmployeeId] = useState('');
  const [password, setPassword] = useState('');
  const [errorMsg, setErrorMsg] = useState('');

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    const trimmedId = employeeId.trim();
    const matched = users.find(
      (u) => u.employeeId === trimmedId && u.password === password
    );

    if (!matched) {
      setErrorMsg('รหัสพนักงานหรือรหัสผ่านไม่ถูกต้อง กรุณากรอกใหม่อีกครั้ง');
      return;
    }

    if (currentUser.employeeId && matched.employeeId !== currentUser.employeeId) {
      setErrorMsg(
        `กรุณากรอกรหัสพนักงานและรหัสผ่านของบัญชีตนเอง (${currentUser.employeeId}) เพื่อยืนยันตัวตน`
      );
      return;
    }

    setErrorMsg('');
    onVerified(matched);
  };

  return (
    <div
      className="fixed inset-0 z-50 bg-rose-950/40 backdrop-blur-xs flex items-center justify-center p-4"
      role="dialog"
      aria-modal="true"
      aria-labelledby="verify-modal-title"
    >
      <div className="bg-white border border-rose-200 rounded-xl max-w-md w-full p-6 space-y-5 shadow-xl">
        <div className="flex items-start justify-between gap-4 border-b border-rose-100 pb-4">
          <div className="flex items-center gap-2.5">
            <div className="w-9 h-9 rounded-lg bg-pink-50 text-pink-600 flex items-center justify-center shrink-0 border border-pink-100">
              <ShieldCheck size={19} aria-hidden="true" />
            </div>
            <div>
              <h2 id="verify-modal-title" className="text-base font-bold text-slate-900">
                ยืนยันตัวตนพนักงานก่อนดำเนินการ
              </h2>
              <p className="text-xs text-slate-500 mt-0.5">
                กรุณากรอกรหัสพนักงานและรหัสผ่านของตนเพื่อยืนยันการทำรายการ
              </p>
            </div>
          </div>
          <button
            type="button"
            onClick={onCancel}
            aria-label="ปิดหน้าต่างยืนยันตัวตน"
            className="p-1 text-slate-400 hover:text-rose-700 rounded-lg cursor-pointer"
          >
            <X size={18} />
          </button>
        </div>

        <div className="bg-rose-50/70 border border-rose-100 rounded-lg p-3.5 space-y-1 text-xs">
          <p className="text-slate-500">กิจกรรมที่กำลังจะดำเนินการ</p>
          <p className="font-semibold text-slate-900">{actionLabel}</p>
          <p className="text-slate-500 font-mono pt-1">
            บัญชีที่เข้าสู่ระบบ: {currentUser.name}
            {currentUser.employeeId ? ` (${currentUser.employeeId})` : ''}
          </p>
        </div>

        {errorMsg && (
          <div
            role="alert"
            className="p-3 rounded-lg bg-rose-50 border border-rose-200 text-rose-800 text-xs leading-relaxed"
          >
            {errorMsg}
          </div>
        )}

        <form onSubmit={handleSubmit} className="space-y-4">
          <div className="space-y-1.5">
            <label htmlFor="verify-emp-id" className="block text-xs font-medium text-slate-700">
              รหัสพนักงาน (Employee ID) *
            </label>
            <input
              id="verify-emp-id"
              type="text"
              required
              autoFocus
              value={employeeId}
              onChange={(e) => setEmployeeId(e.target.value)}
              placeholder="กรอกรหัสพนักงานของคุณ"
              className="w-full px-3 py-2 text-sm border border-rose-200 rounded-lg font-mono focus:border-pink-500"
            />
          </div>

          <div className="space-y-1.5">
            <label htmlFor="verify-password" className="block text-xs font-medium text-slate-700">
              รหัสผ่าน (Password) *
            </label>
            <input
              id="verify-password"
              type="password"
              required
              value={password}
              onChange={(e) => setPassword(e.target.value)}
              placeholder="••••••"
              className="w-full px-3 py-2 text-sm border border-rose-200 rounded-lg font-mono focus:border-pink-500"
            />
          </div>

          <div className="flex items-center justify-end gap-2 pt-2">
            <button
              type="button"
              onClick={onCancel}
              className="px-4 py-2 bg-rose-50 hover:bg-rose-100 text-rose-700 text-xs font-medium rounded-lg transition-colors cursor-pointer"
            >
              ยกเลิก
            </button>
            <button
              type="submit"
              className="px-4 py-2 bg-pink-600 hover:bg-pink-700 text-white text-xs font-semibold rounded-lg transition-colors cursor-pointer"
            >
              ยืนยันตัวตนและทำรายการ
            </button>
          </div>
        </form>
      </div>
    </div>
  );
}

import React, { useMemo, useState } from 'react';
import { DollarSign, Download, Edit3, Save, TrendingUp } from 'lucide-react';
import {
  Bar,
  BarChart,
  CartesianGrid,
  Legend,
  ResponsiveContainer,
  Tooltip,
  XAxis,
  YAxis,
} from 'recharts';
import {
  CostBreakdown,
  formatCurrency,
  getTotalRequestCost,
  ReturnRequest,
} from '../types';
import { StatusIndicator } from './StatusText';

interface CostManagementViewProps {
  requests: ReturnRequest[];
  onUpdateCost: (requestId: string, nextCosts: CostBreakdown, onSuccess: () => void) => void;
  onExportCSV: () => void;
}

export function CostManagementView({
  requests,
  onUpdateCost,
  onExportCSV,
}: CostManagementViewProps) {
  const [editingId, setEditingId] = useState<string | null>(null);
  const [draftCost, setDraftCost] = useState<CostBreakdown>({
    freight: 0,
    inspectionFee: 0,
    repair: 0,
    assetLoss: 0,
  });

  const totals = useMemo(() => {
    return requests.reduce(
      (acc, r) => {
        acc.freight += r.costs.freight;
        acc.inspectionFee += r.costs.inspectionFee;
        acc.repair += r.costs.repair;
        acc.assetLoss += r.costs.assetLoss;
        acc.grandTotal += getTotalRequestCost(r);
        acc.returnValue += r.unitPrice * r.qty;
        return acc;
      },
      {
        freight: 0,
        inspectionFee: 0,
        repair: 0,
        assetLoss: 0,
        grandTotal: 0,
        returnValue: 0,
      }
    );
  }, [requests]);

  const costRatio =
    totals.returnValue > 0
      ? ((totals.grandTotal / totals.returnValue) * 100).toFixed(1)
      : '0.0';

  const chartData = useMemo(() => {
    return requests.map((r) => ({
      id: r.id.slice(-4),
      fullId: r.id,
      ค่าขนส่ง: r.costs.freight,
      ค่าตรวจสอบ: r.costs.inspectionFee,
      ค่าซ่อมแซม: r.costs.repair,
      มูลค่าเสียหาย: r.costs.assetLoss,
    }));
  }, [requests]);

  const startEditing = (req: ReturnRequest) => {
    setEditingId(req.id);
    setDraftCost({ ...req.costs });
  };

  const handleSave = (id: string) => {
    onUpdateCost(
      id,
      {
        freight: Math.max(0, draftCost.freight),
        inspectionFee: Math.max(0, draftCost.inspectionFee),
        repair: Math.max(0, draftCost.repair),
        assetLoss: Math.max(0, draftCost.assetLoss),
      },
      () => setEditingId(null)
    );
  };

  return (
    <div className="space-y-8">
      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
        <div>
          <h1 className="text-2xl font-bold text-slate-900">
            บริหารและวิเคราะห์ต้นทุนคืนสินค้า (Cost Analytics)
          </h1>
          <p className="text-sm text-slate-500 mt-1">
            ตรวจสอบโครงสร้างค่าใช้จ่าย Reverse Logistics แยกตามรายการและปรับปรุงตัวเลขทางบัญชี
          </p>
        </div>

        <button
          type="button"
          onClick={onExportCSV}
          className="inline-flex items-center justify-center gap-2 px-4 py-2 bg-pink-600 hover:bg-pink-700 text-white text-sm font-medium rounded-lg shadow-xs shadow-pink-500/15 transition-colors whitespace-nowrap shrink-0 cursor-pointer"
        >
          <Download size={16} aria-hidden="true" />
          <span>ส่งออกรายงานต้นทุน CSV</span>
        </button>
      </div>

      {/* Cost KPI Strip */}
      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-5 gap-4">
        <div className="bg-white border border-rose-100 rounded-xl p-5 space-y-1.5 shadow-xs shadow-pink-500/5">
          <p className="text-xs font-medium text-slate-500">ต้นทุนรวมทั้งหมด</p>
          <p className="text-2xl font-bold font-mono tabular-nums text-slate-900">
            {formatCurrency(totals.grandTotal)}
          </p>
          <p className="text-xs text-slate-500 font-mono tabular-nums flex items-center gap-1">
            <TrendingUp size={12} className="text-pink-600" aria-hidden="true" />
            <span>{costRatio}% ของมูลค่าสินค้าคืน</span>
          </p>
        </div>

        <div className="bg-white border border-rose-100 rounded-xl p-5 space-y-1.5 shadow-xs shadow-pink-500/5">
          <p className="text-xs font-medium text-slate-500">ค่าขนส่งขากลับ (Freight)</p>
          <p className="text-2xl font-bold font-mono tabular-nums text-pink-700">
            {formatCurrency(totals.freight)}
          </p>
          <p className="text-xs text-slate-500">ค่าบริการรับ-ส่งพัสดุคืน</p>
        </div>

        <div className="bg-white border border-rose-100 rounded-xl p-5 space-y-1.5 shadow-xs shadow-pink-500/5">
          <p className="text-xs font-medium text-slate-500">ค่าตรวจสอบ (Inspection)</p>
          <p className="text-2xl font-bold font-mono tabular-nums text-fuchsia-700">
            {formatCurrency(totals.inspectionFee)}
          </p>
          <p className="text-xs text-slate-500">ค่าแรงและอุปกรณ์ทดสอบ</p>
        </div>

        <div className="bg-white border border-rose-100 rounded-xl p-5 space-y-1.5 shadow-xs shadow-pink-500/5">
          <p className="text-xs font-medium text-slate-500">ค่าซ่อมแซม (Repair)</p>
          <p className="text-2xl font-bold font-mono tabular-nums text-amber-700">
            {formatCurrency(totals.repair)}
          </p>
          <p className="text-xs text-slate-500">ค่าอะไหล่และซ่อมบำรุง</p>
        </div>

        <div className="bg-white border border-rose-100 rounded-xl p-5 space-y-1.5 shadow-xs shadow-pink-500/5">
          <p className="text-xs font-medium text-slate-500">ความเสียหายสุทธิ (Asset Loss)</p>
          <p className="text-2xl font-bold font-mono tabular-nums text-rose-700">
            {formatCurrency(totals.assetLoss)}
          </p>
          <p className="text-xs text-slate-500">สินค้าตัดจำหน่ายซาก</p>
        </div>
      </div>

      {/* Stacked Cost Breakdown Chart */}
      <div className="bg-white border border-rose-100 rounded-xl p-6 space-y-4">
        <div className="flex items-center justify-between">
          <div>
            <h2 className="text-base font-semibold text-slate-900">
              สัดส่วนต้นทุนรายคำร้อง (Stacked Cost Breakdown)
            </h2>
            <p className="text-xs text-slate-500 mt-0.5">
              แสดงรหัส 4 หลักท้ายของคำร้อง RTX พร้อมองค์ประกอบค่าใช้จ่ายทั้ง 4 หมวด
            </p>
          </div>
          <DollarSign size={18} className="text-pink-500" aria-hidden="true" />
        </div>

        <div className="h-72 w-full">
          <ResponsiveContainer width="100%" height="100%">
            <BarChart data={chartData} margin={{ top: 10, right: 16, left: 0, bottom: 8 }}>
              <CartesianGrid strokeDasharray="3 3" stroke="#fce7f3" vertical={false} />
              <XAxis dataKey="id" tick={{ fontSize: 12, fill: '#64748b' }} />
              <YAxis tick={{ fontSize: 12, fill: '#64748b' }} />
              <Tooltip
                formatter={(value, name) => [
                  `฿${Number(value ?? 0).toLocaleString()}`,
                  String(name ?? ''),
                ]}
                contentStyle={{
                  borderRadius: '8px',
                  border: '1px solid #fbcfe8',
                  fontSize: '12px',
                }}
              />
              <Legend wrapperStyle={{ fontSize: '12px' }} />
              <Bar dataKey="ค่าขนส่ง" stackId="a" fill="#ec4899" />
              <Bar dataKey="ค่าตรวจสอบ" stackId="a" fill="#be185d" />
              <Bar dataKey="ค่าซ่อมแซม" stackId="a" fill="#f59e0b" />
              <Bar dataKey="มูลค่าเสียหาย" stackId="a" fill="#dc2626" radius={[4, 4, 0, 0]} />
            </BarChart>
          </ResponsiveContainer>
        </div>
      </div>

      {/* Editable Cost Ledger */}
      <div className="bg-white border border-rose-100 rounded-xl overflow-hidden">
        <div className="px-6 py-4 border-b border-rose-100 flex items-center justify-between">
          <h2 className="text-base font-semibold text-slate-900">
            ตารางบันทึกและปรับปรุงต้นทุนรายรายการ
          </h2>
          <span className="text-xs text-slate-500">คลิก "แก้ไข" เพื่อปรับตัวเลขต้นทุน</span>
        </div>

        <div className="overflow-x-auto">
          <table className="w-full text-sm text-left">
            <thead className="bg-rose-50/70 text-xs text-rose-900/70 border-b border-rose-100">
              <tr>
                <th className="py-3 px-4 font-medium">รหัสคำร้อง</th>
                <th className="py-3 px-4 font-medium">สินค้า &amp; สถานะ</th>
                <th className="py-3 px-4 font-medium text-right">ค่าขนส่ง</th>
                <th className="py-3 px-4 font-medium text-right">ค่าตรวจสอบ</th>
                <th className="py-3 px-4 font-medium text-right">ค่าซ่อม</th>
                <th className="py-3 px-4 font-medium text-right">ความเสียหาย</th>
                <th className="py-3 px-4 font-medium text-right">รวมต้นทุน</th>
                <th className="py-3 px-4 font-medium text-right">จัดการ</th>
              </tr>
            </thead>
            <tbody className="divide-y divide-rose-100/70">
              {requests.map((req) => {
                const isEditing = editingId === req.id;
                const rowTotal = isEditing
                  ? draftCost.freight +
                    draftCost.inspectionFee +
                    draftCost.repair +
                    draftCost.assetLoss
                  : getTotalRequestCost(req);

                return (
                  <tr key={req.id} className="hover:bg-rose-50/40">
                    <td className="py-3 px-4 font-mono text-xs font-semibold text-slate-900 whitespace-nowrap">
                      {req.id}
                    </td>
                    <td className="py-3 px-4">
                      <div className="font-medium text-slate-800 truncate max-w-[180px]">
                        {req.productName}
                      </div>
                      <div className="mt-1">
                        <StatusIndicator status={req.status} />
                      </div>
                    </td>
                    <td className="py-3 px-4 text-right font-mono text-xs tabular-nums">
                      {isEditing ? (
                        <input
                          type="number"
                          min={0}
                          value={draftCost.freight}
                          onChange={(e) =>
                            setDraftCost((p) => ({ ...p, freight: Number(e.target.value) || 0 }))
                          }
                          className="w-20 px-2 py-1 text-right border border-pink-400 rounded"
                        />
                      ) : (
                        formatCurrency(req.costs.freight)
                      )}
                    </td>
                    <td className="py-3 px-4 text-right font-mono text-xs tabular-nums">
                      {isEditing ? (
                        <input
                          type="number"
                          min={0}
                          value={draftCost.inspectionFee}
                          onChange={(e) =>
                            setDraftCost((p) => ({
                              ...p,
                              inspectionFee: Number(e.target.value) || 0,
                            }))
                          }
                          className="w-20 px-2 py-1 text-right border border-pink-400 rounded"
                        />
                      ) : (
                        formatCurrency(req.costs.inspectionFee)
                      )}
                    </td>
                    <td className="py-3 px-4 text-right font-mono text-xs tabular-nums">
                      {isEditing ? (
                        <input
                          type="number"
                          min={0}
                          value={draftCost.repair}
                          onChange={(e) =>
                            setDraftCost((p) => ({ ...p, repair: Number(e.target.value) || 0 }))
                          }
                          className="w-20 px-2 py-1 text-right border border-pink-400 rounded"
                        />
                      ) : (
                        formatCurrency(req.costs.repair)
                      )}
                    </td>
                    <td className="py-3 px-4 text-right font-mono text-xs tabular-nums">
                      {isEditing ? (
                        <input
                          type="number"
                          min={0}
                          value={draftCost.assetLoss}
                          onChange={(e) =>
                            setDraftCost((p) => ({
                              ...p,
                              assetLoss: Number(e.target.value) || 0,
                            }))
                          }
                          className="w-24 px-2 py-1 text-right border border-pink-400 rounded"
                        />
                      ) : (
                        formatCurrency(req.costs.assetLoss)
                      )}
                    </td>
                    <td className="py-3 px-4 text-right font-mono text-xs font-bold tabular-nums text-slate-900">
                      {formatCurrency(rowTotal)}
                    </td>
                    <td className="py-3 px-4 text-right whitespace-nowrap">
                      {isEditing ? (
                        <button
                          type="button"
                          onClick={() => handleSave(req.id)}
                          className="inline-flex items-center gap-1 px-2.5 py-1.5 bg-emerald-600 hover:bg-emerald-700 text-white text-xs font-medium rounded-md cursor-pointer"
                        >
                          <Save size={13} aria-hidden="true" />
                          <span>บันทึก</span>
                        </button>
                      ) : (
                        <button
                          type="button"
                          onClick={() => startEditing(req)}
                          className="inline-flex items-center gap-1 px-2.5 py-1.5 bg-rose-50 hover:bg-rose-100 text-rose-700 border border-rose-200/70 text-xs font-medium rounded-md cursor-pointer"
                        >
                          <Edit3 size={13} aria-hidden="true" />
                          <span>แก้ไข</span>
                        </button>
                      )}
                    </td>
                  </tr>
                );
              })}
            </tbody>
          </table>
        </div>
      </div>
    </div>
  );
}

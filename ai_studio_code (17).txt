import React, { useState } from 'react';
import { PlusCircle, Shield, Trash2, Users } from 'lucide-react';
import { Role, StaffUser } from '../types';

interface UserManagementViewProps {
  users: StaffUser[];
  onCreateUser: (newUser: Omit<StaffUser, 'level'>, onSuccess: () => void) => string | null;
  onDeleteUser: (employeeId: string) => void;
}

export function UserManagementView({
  users,
  onCreateUser,
  onDeleteUser,
}: UserManagementViewProps) {
  const [employeeId, setEmployeeId] = useState('');
  const [name, setName] = useState('');
  const [password, setPassword] = useState('');
  const [role, setRole] = useState<Exclude<Role, 'Customer'>>('Warehouse Staff');
  const [errorMsg, setErrorMsg] = useState('');

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (!employeeId.trim() || !name.trim() || !password.trim()) {
      setErrorMsg('กรุณากรอกรหัสพนักงาน ชื่อ-นามสกุล และรหัสผ่านให้ครบถ้วน');
      return;
    }

    const err = onCreateUser(
      {
        employeeId: employeeId.trim(),
        name: name.trim(),
        password: password.trim(),
        role,
      },
      () => {
        setEmployeeId('');
        setName('');
        setPassword('');
        setRole('Warehouse Staff');
      }
    );

    if (err) {
      setErrorMsg(err);
    } else {
      setErrorMsg('');
    }
  };

  return (
    <div className="space-y-8 max-w-5xl">
      <div>
        <h1 className="text-2xl font-bold text-slate-900">
          จัดการผู้ใช้งานในระบบ (Super Admin 000001)
        </h1>
        <p className="text-sm text-slate-500 mt-1">
          เพิ่มและกำหนดรหัสพนักงาน รหัสผ่าน และบทบาทหน้าที่สำหรับ Admin, พนักงานคลัง และผู้ตรวจสอบสินค้า
        </p>
      </div>

      {/* Create Employee Form */}
      <form
        onSubmit={handleSubmit}
        className="bg-white border border-rose-100 rounded-xl p-6 space-y-5"
      >
        <div className="flex items-center gap-2 border-b border-rose-100 pb-3">
          <PlusCircle size={17} className="text-pink-600" aria-hidden="true" />
          <h2 className="text-base font-semibold text-slate-900">สร้างบัญชีพนักงานใหม่</h2>
        </div>

        {errorMsg && (
          <div
            role="alert"
            className="p-3 rounded-lg bg-rose-50 border border-rose-200 text-rose-800 text-xs"
          >
            {errorMsg}
          </div>
        )}

        <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
          <div className="space-y-1.5">
            <label htmlFor="emp-id" className="block text-xs font-medium text-slate-700">
              รหัสพนักงาน (Employee ID) *
            </label>
            <input
              id="emp-id"
              type="text"
              required
              value={employeeId}
              onChange={(e) => setEmployeeId(e.target.value)}
              placeholder="เช่น WH002 หรือ INS002"
              className="w-full px-3 py-2 text-sm border border-rose-200 rounded-lg font-mono focus:border-pink-500"
            />
          </div>

          <div className="space-y-1.5">
            <label htmlFor="emp-name" className="block text-xs font-medium text-slate-700">
              ชื่อ-นามสกุล *
            </label>
            <input
              id="emp-name"
              type="text"
              required
              value={name}
              onChange={(e) => setName(e.target.value)}
              placeholder="เช่น ณัฐพล จัดส่ง"
              className="w-full px-3 py-2 text-sm border border-rose-200 rounded-lg focus:border-pink-500"
            />
          </div>

          <div className="space-y-1.5">
            <label htmlFor="emp-pass" className="block text-xs font-medium text-slate-700">
              รหัสผ่านเริ่มต้น *
            </label>
            <input
              id="emp-pass"
              type="text"
              required
              value={password}
              onChange={(e) => setPassword(e.target.value)}
              placeholder="กำหนดรหัสผ่าน"
              className="w-full px-3 py-2 text-sm border border-rose-200 rounded-lg font-mono focus:border-pink-500"
            />
          </div>

          <div className="space-y-1.5">
            <label htmlFor="emp-role" className="block text-xs font-medium text-slate-700">
              บทบาทหน้าที่ (Role) *
            </label>
            <select
              id="emp-role"
              value={role}
              onChange={(e) => setRole(e.target.value as Exclude<Role, 'Customer'>)}
              className="w-full px-3 py-2 text-sm border border-rose-200 rounded-lg bg-white focus:border-pink-500"
            >
              <option value="Admin">Admin (ผู้ดูแลระบบ)</option>
              <option value="Warehouse Staff">Warehouse Staff (พนักงานคลัง)</option>
              <option value="Inspector">Inspector (ผู้ตรวจสอบสินค้า)</option>
            </select>
          </div>
        </div>

        <div className="flex justify-end pt-2">
          <button
            type="submit"
            className="inline-flex items-center gap-2 px-5 py-2.5 bg-gradient-to-r from-pink-500 to-rose-500 hover:from-pink-600 hover:to-rose-600 text-white text-sm font-semibold rounded-lg shadow-sm shadow-pink-500/20 transition-all cursor-pointer"
          >
            <PlusCircle size={16} aria-hidden="true" />
            <span>บันทึกสร้างบัญชีพนักงานใหม่</span>
          </button>
        </div>
      </form>

      {/* Staff Directory Table */}
      <div className="bg-white border border-rose-100 rounded-xl overflow-hidden">
        <div className="px-6 py-4 border-b border-rose-100 flex items-center justify-between">
          <div className="flex items-center gap-2">
            <Users size={17} className="text-pink-600" aria-hidden="true" />
            <h2 className="text-base font-semibold text-slate-900">
              รายชื่อพนักงานทั้งหมดในระบบ
            </h2>
          </div>
          <span className="text-xs font-mono text-slate-500 tabular-nums">
            {users.length} บัญชี
          </span>
        </div>

        <div className="overflow-x-auto">
          <table className="w-full text-sm text-left">
            <thead className="bg-rose-50/70 text-xs text-rose-900/70 border-b border-rose-100">
              <tr>
                <th className="py-3 px-4 font-medium">รหัสพนักงาน</th>
                <th className="py-3 px-4 font-medium">ชื่อ-นามสกุล</th>
                <th className="py-3 px-4 font-medium">บทบาทหน้าที่</th>
                <th className="py-3 px-4 font-medium">ระดับสิทธิ์</th>
                <th className="py-3 px-4 font-medium">รหัสผ่าน</th>
                <th className="py-3 px-4 font-medium text-right">จัดการ</th>
              </tr>
            </thead>
            <tbody className="divide-y divide-rose-100/70">
              {users.map((u) => (
                <tr key={u.employeeId} className="hover:bg-rose-50/40">
                  <td className="py-3 px-4 font-mono text-xs font-semibold text-pink-700 whitespace-nowrap">
                    {u.employeeId}
                  </td>
                  <td className="py-3 px-4 font-medium text-slate-900">{u.name}</td>
                  <td className="py-3 px-4 text-xs font-medium text-slate-700">{u.role}</td>
                  <td className="py-3 px-4 text-xs font-mono text-slate-600">
                    {u.level === 1 ? (
                      <span className="inline-flex items-center gap-1 text-rose-700 font-semibold">
                        <Shield size={13} aria-hidden="true" />
                        <span>Level 1 (Super Admin)</span>
                      </span>
                    ) : (
                      <span>Level 2 (Staff)</span>
                    )}
                  </td>
                  <td className="py-3 px-4 font-mono text-xs text-slate-500">
                    {u.password}
                  </td>
                  <td className="py-3 px-4 text-right">
                    {u.level === 1 ? (
                      <span className="text-xs text-slate-400">บัญชีหลักระบบ</span>
                    ) : (
                      <button
                        type="button"
                        onClick={() => onDeleteUser(u.employeeId)}
                        className="inline-flex items-center gap-1 px-2.5 py-1.5 text-xs font-medium text-rose-700 hover:bg-rose-50 rounded-md transition-colors cursor-pointer"
                      >
                        <Trash2 size={13} aria-hidden="true" />
                        <span>ลบบัญชี</span>
                      </button>
                    )}
                  </td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      </div>
    </div>
  );
}

function formatNow(): string {
  const now = new Date();
  const y = now.getFullYear();
  const m = String(now.getMonth() + 1).padStart(2, '0');
  const d = String(now.getDate()).padStart(2, '0');
  const hh = String(now.getHours()).padStart(2, '0');
  const mm = String(now.getMinutes()).padStart(2, '0');
  return `${y}-${m}-${d} ${hh}:${mm}`;
}

function loadStored<T>(key: string, fallback: T): T {
  try {
    const raw = window.localStorage.getItem(key);
    if (!raw) return fallback;
    return JSON.parse(raw) as T;
  } catch {
    return fallback;
  }
}

                customerEmail: payload.sentToEmail,
                status: nextStatus,
                refundSlip: slipRecord,
              }
            : prev
        );
        addNotification(
          `${actor} ส่งสลิปโอนเงินคืนไปยังอีเมล ${payload.sentToEmail} สำหรับคำร้อง ${requestId}`,
          'success',
          requestId
        );
        showToast(`ส่งสลิปโอนเงินคืนไปยังอีเมล ${payload.sentToEmail} เรียบร้อยแล้ว`);
        onSuccess();
      }
    );
  };

  const handleUpdateStatus = (id: string, nextStatus: ReturnStatus) => {
    const requiredRole = STEP_APPROVER_ROLE[nextStatus];
    if (requiredRole && currentUser?.role !== requiredRole) {
      showToast(
        `ขั้นตอน "${nextStatus}" ต้องได้รับการอนุมัติจาก${ROLE_THAI_LABELS[requiredRole]}เท่านั้น`,
        'error'
      );
      return;
    }

    const targetReq = requests.find((r) => r.id === id);
    if (
      nextStatus === 'ตรวจสอบเสร็จสิ้น / ปิดรายการ' &&
      targetReq?.resolution?.type === 'refund' &&
      (!targetReq.refundSlip?.transferRef?.trim() || !targetReq.refundSlip?.slipImageUrl)
    ) {
      showToast('ขั้นตอนคืนเงิน ต้องส่งสลิปโอนเงินและกรอกเลขอ้างอิงก่อน จึงจะไปขั้นต่อไปได้', 'error');
      return;
    }

    requestStaffVerification(
      `อนุมัติขั้นตอน "${nextStatus}" สำหรับคำร้อง ${id}`,
      (verifiedUser) => {
        if (requiredRole && verifiedUser.role !== requiredRole) {
          showToast(
            `ขั้นตอน "${nextStatus}" ต้องได้รับการอนุมัติจาก${ROLE_THAI_LABELS[requiredRole]}เท่านั้น`,
            'error'
          );
          return;
        }
        const actor = `${verifiedUser.name} (${verifiedUser.employeeId})`;
        const timestamp = formatNow();
        const note =
          nextStatus === 'ปฏิเสธคำขอคืนสินค้า'
            ? 'ปฏิเสธคำขอคืนสินค้า'
            : `อนุมัติขั้นตอน: ${nextStatus}`;

        setRequests((prev) =>
          prev.map((r) =>
            r.id === id
              ? {
                  ...r,
                  status: nextStatus,
                  timeline: [
                    ...r.timeline,
                    { status: nextStatus, date: timestamp, actor, note },
                  ],
                }
              : r
          )
        );
        setModalRequest((prev) =>
          prev && prev.id === id
            ? {
                ...prev,
                status: nextStatus,
                timeline: [
                  ...prev.timeline,
                  { status: nextStatus, date: timestamp, actor, note },
                ],
              }
            : prev
        );
        addNotification(`อนุมัติสถานะ ${id} เป็น "${nextStatus}" โดย ${actor}`, 'info', id);
        showToast(`อนุมัติขั้นตอน "${nextStatus}" สำหรับ ${id} เรียบร้อยแล้ว`);
      }
    );
  };

  const handleReceiveWarehouse = (
    requestId: string,
    warehouseId: string,
    locationBin: string
  ) => {
    if (currentUser?.role !== 'Warehouse Staff') {
      showToast('ขั้นตอน "ได้รับสินค้าแล้ว" ต้องได้รับการอนุมัติจากพนักงานคลังเท่านั้น', 'error');
      return;
    }

    const cleanBin = locationBin.trim() || 'RACK-A01';
    requestStaffVerification(
      `อนุมัติขั้นตอน "ได้รับสินค้าแล้ว" · รับพัสดุ ${requestId} เข้าคลัง ${warehouseId} (${cleanBin})`,
      (verifiedUser) => {
        if (verifiedUser.role !== 'Warehouse Staff') {
          showToast('ขั้นตอน "ได้รับสินค้าแล้ว" ต้องได้รับการอนุมัติจากพนักงานคลังเท่านั้น', 'error');
          return;
        }
        const nowStr = formatNow();
        const actor = `${verifiedUser.name} (${verifiedUser.employeeId})`;
        const targetReq = requests.find((r) => r.id === requestId);
        const qtyToAdd = targetReq?.qty || 1;

        setRequests((prev) =>
          prev.map((r) =>
            r.id === requestId
              ? {
                  ...r,
                  warehouseId,
                  locationBin: cleanBin,
                  receivedDate: nowStr,
                  status: 'ได้รับสินค้าแล้ว',
                  timeline: [
                    ...r.timeline,
                    {
                      status: 'ได้รับสินค้าแล้ว',
                      date: nowStr,
                      actor,
                      note: `พนักงานคลังอนุมัติได้รับสินค้าแล้ว · รับเข้าคลัง ${warehouseId} ช่อง ${cleanBin}`,
                    },
                  ],
                }
              : r
          )
        );

        setWarehouses((prev) =>
          prev.map((w) =>
            w.id === warehouseId
              ? { ...w, occupied: Math.min(w.capacity, w.occupied + qtyToAdd) }
              : w
          )
        );

        addNotification(
          `คลัง ${warehouseId} อนุมัติได้รับสินค้าแล้ว (${requestId}) ช่อง ${cleanBin} โดย ${actor}`,
          'success',
          requestId
        );
        showToast(`อนุมัติได้รับสินค้าแล้ว (${requestId}) เข้าคลัง ${warehouseId} สำเร็จ`);
      }
    );
  };

  const handleDispatchToCustomer = (
    requestId: string,
    trackingNumber: string,
    productName: string,
    carrier: string,
    closeAfterDispatch: boolean,
    onSuccess: () => void,
    productSku?: string
  ) => {
    requestStaffVerification(
      `ส่งพัสดุถึงลูกค้า (${requestId}) · เลขพัสดุ ${trackingNumber} · สินค้า ${productName}`,
      (verifiedUser) => {
        const nowStr = formatNow();
        const actor = `${verifiedUser.name} (${verifiedUser.employeeId})`;
        const targetReq = requests.find((r) => r.id === requestId);
        const qtyToDeduct = targetReq?.qty || 1;
        const canAdminChangeStatus = verifiedUser.role === 'Admin';
        const nextStatus: ReturnStatus = canAdminChangeStatus
          ? closeAfterDispatch
            ? 'ตรวจสอบเสร็จสิ้น / ปิดรายการ'
            : 'คืนเงิน / เปลี่ยนสินค้า'
          : targetReq?.status || 'คืนเงิน / เปลี่ยนสินค้า';

        const dispatchRecord = {
          trackingNumber,
          productName,
          carrier,
          shippedAt: nowStr,
          shippedBy: actor,
        };

        setRequests((prev) =>
          prev.map((r) =>
            r.id === requestId
              ? {
                  ...r,
                  status: nextStatus,
                  outboundDispatch: dispatchRecord,
                  timeline: [
                    ...r.timeline,
                    {
                      status: nextStatus,
                      date: nowStr,
                      actor,
                      note:
                        `จัดส่งพัสดุถึงลูกค้า (${carrier}) · เลขพัสดุ ${trackingNumber} · สินค้า: ${productName}` +
                        (!canAdminChangeStatus
                          ? ' (รอผู้ดูแลและแอดมินอนุมัติปิดรายการ)'
                          : ''),
                    },
                  ],
                }
              : r
          )
        );

        if (productSku) {
          setProducts((prev) =>
            prev.map((p) =>
              p.sku === productSku
                ? { ...p, stock: Math.max(0, p.stock - qtyToDeduct) }
                : p
            )
          );
        }

        addNotification(
          `${actor} ส่งพัสดุ ${trackingNumber} (${productName}) ให้ลูกค้าคำร้อง ${requestId}`,
          'success',
          requestId
        );
        showToast(`บันทึกการจัดส่งพัสดุ ${trackingNumber} ถึงลูกค้าสำเร็จ`);
        onSuccess();
      }
    );
  };

  const handleCompleteInspection = (
    requestId: string,
    record: InspectionRecord
  ) => {
    if (currentUser?.role !== 'Inspector') {
      showToast('เฉพาะฝ่ายตรวจสอบเท่านั้นที่สามารถบันทึกผลการตรวจสอบสภาพสินค้าได้', 'error');
      return;
    }

    requestStaffVerification(
      `บันทึกผลตรวจสภาพสินค้า ${requestId} (${record.disposition})`,
      (verifiedUser) => {
        if (verifiedUser.role !== 'Inspector') {
          showToast('เฉพาะฝ่ายตรวจสอบเท่านั้นที่สามารถบันทึกผลการตรวจสอบสภาพสินค้าได้', 'error');
          return;
        }
        const targetReq = requests.find((r) => r.id === requestId);
        const nextStatus: ReturnStatus = 'อยู่ระหว่างตรวจสอบสินค้า';
        const timestamp = formatNow();
        const actor = `${verifiedUser.name} (${verifiedUser.employeeId})`;

        setRequests((prev) =>
          prev.map((r) => {
            if (r.id !== requestId) return r;
            return {
              ...r,
              status: nextStatus,
              inspection: {
                ...record,
                inspectorName: actor,
              },
              costs: {
                ...r.costs,
                repair: record.repairCost,
                assetLoss:
                  record.disposition === 'ตัดจำหน่ายเป็นสินค้าเสียหาย'
                    ? record.damagedValue
                    : r.costs.assetLoss,
              },
              timeline: [
                ...r.timeline,
                {
                  status: nextStatus,
                  date: timestamp,
                  actor,
                  note: `${record.grade} · ${record.disposition} (บันทึกผลตรวจแล้ว · รอผู้ดูแล/แอดมินอนุมัติคืนเงิน / เปลี่ยนสินค้า)`,
                },
              ],
            };
          })
        );

        if (targetReq && record.disposition === 'นำกลับเข้าสต็อก') {
          setProducts((prev) =>
            prev.map((p) =>
              p.sku === targetReq.sku ? { ...p, stock: p.stock + targetReq.qty } : p
            )
          );
        }

        addNotification(
          `ตรวจสภาพ ${requestId} เสร็จสิ้น (${record.disposition}) โดย ${actor}`,
          'success',
          requestId
        );
        showToast(`บันทึกผลตรวจสภาพ ${requestId} เรียบร้อยแล้ว`);
      }
    );
  };

  const handleUpdateCost = (
    requestId: string,
    nextCosts: CostBreakdown,
    onSuccess: () => void
  ) => {
    requestStaffVerification(
      `แก้ไขและบันทึกต้นทุนคำร้อง ${requestId}`,
      (verifiedUser) => {
        setRequests((prev) =>
          prev.map((r) => (r.id === requestId ? { ...r, costs: nextCosts } : r))
        );
        addNotification(
          `${verifiedUser.name} (${verifiedUser.employeeId}) อัปเดตต้นทุน ${requestId}`,
          'info',
          requestId
        );
        showToast(`อัปเดตข้อมูลต้นทุนสำหรับ ${requestId} เรียบร้อยแล้ว`);
        onSuccess();
      }
    );
  };

  const handleExportCSV = () => {
    requestStaffVerification('ส่งออกรายงานข้อมูลคำร้องคืนสินค้า (CSV)', (verifiedUser) => {
      const headers = [
        'Request ID',
        'Order Number',
        'Customer',
        'SKU',
        'Product',
        'Qty',
        'Unit Price',
        'Reason',
        'Status',
        'Warehouse',
        'Bin',
        'Freight Cost',
        'Inspection Fee',
        'Repair Cost',
        'Asset Loss',
        'Total Cost',
      ];
      const rows = requests.map((r) => [
        r.id,
        r.orderNumber,
        `"${r.customerName.replace(/"/g, '""')}"`,
        r.sku,
        `"${r.productName.replace(/"/g, '""')}"`,
        r.qty,
        r.unitPrice,
        `"${r.reason.replace(/"/g, '""')}"`,
        `"${r.status}"`,
        r.warehouseId,
        r.locationBin,
        r.costs.freight,
        r.costs.inspectionFee,
        r.costs.repair,
        r.costs.assetLoss,
        getTotalRequestCost(r),
      ]);

      const csvContent =
        '\uFEFF' + [headers.join(','), ...rows.map((row) => row.join(','))].join('\n');
      const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
      const url = URL.createObjectURL(blob);
      const link = document.createElement('a');
      link.href = url;
      link.download = `routex-returns-${new Date().toISOString().slice(0, 10)}.csv`;
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
      URL.revokeObjectURL(url);
      addNotification(
        `${verifiedUser.name} (${verifiedUser.employeeId}) ส่งออกรายงาน CSV`,
        'info'
      );
      showToast('ส่งออกไฟล์รายงาน CSV สำเร็จ');
    });
  };

  if (!currentUser) {
    return (
      <AuthPortal
        siteConfig={siteConfig}
        onStaffLogin={handleStaffLogin}
        onCustomerEnter={handleCustomerEnter}
      />
    );
  }

  const activeRole = currentUser.role;
  const navItems = ROLE_NAV_MAP[activeRole].filter(
    (item) => !item.requireLevel1 || currentUser.level === 1
  );
  const selectedRequest =
    activeRole === 'Customer'
      ? requests.find((r) => r.id === selectedRequestId) || null
      : requests.find((r) => r.id === selectedRequestId) || requests[0] || null;
  const activeNavLabel =
    navItems.find((item) => item.id === activeTab)?.label || 'Workspace';

  const visibleCustomDetails = siteConfig.customDetails.filter(
    (d) =>
      d.placement === 'all' ||
      (d.placement === 'customer' && activeRole === 'Customer')
  );

  return (
    <div className="min-h-screen bg-[#fff9fb] text-slate-900 flex flex-col">
      <header className="bg-white/95 backdrop-blur-sm text-slate-900 border-b border-rose-200/80 sticky top-0 z-40">
        <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 h-16 flex items-center justify-between gap-8">
          <div className="flex items-center gap-3 shrink-0">
            <button
              type="button"
              onClick={() => setSidebarOpen((prev) => !prev)}
              aria-label="เปิดหรือปิดเมนูด้านข้าง"
              className="md:hidden text-rose-700 hover:bg-rose-50 p-1.5 rounded-lg cursor-pointer"
            >
              {sidebarOpen ? <X size={22} /> : <Menu size={22} />}
            </button>
            <button
              type="button"
              onClick={() => enterWorkspaceForRole(activeRole)}
              className="flex items-center gap-2.5 text-lg font-bold tracking-tight text-slate-900 whitespace-nowrap shrink-0 cursor-pointer"
            >
              <span className="w-8 h-8 rounded-lg bg-gradient-to-br from-pink-500 to-rose-500 text-white shadow-sm shadow-pink-500/20 flex items-center justify-center">
                <Truck size={17} aria-hidden="true" />
              </span>
              <span>{siteConfig.brandName}</span>
            </button>
          </div>

          <div className="hidden md:flex items-center gap-2 text-sm text-slate-500 truncate">
            <span className="text-slate-900 font-medium whitespace-nowrap">
              {currentUser.name}
            </span>
            <span aria-hidden="true" className="text-rose-300">·</span>
            <span className="whitespace-nowrap">
              {activeRole}
              {currentUser.employeeId ? ` (${currentUser.employeeId})` : ''}
            </span>
            <span aria-hidden="true" className="text-rose-300">/</span>
            <span className="truncate text-pink-700 font-medium">{activeNavLabel}</span>
          </div>

          <div className="flex items-center shrink-0">
            <button
              type="button"
              onClick={handleLogout}
              className="inline-flex items-center gap-2 px-3.5 py-2 bg-rose-50 hover:bg-rose-100 text-rose-700 text-xs font-medium rounded-lg border border-rose-200 transition-colors whitespace-nowrap shrink-0 cursor-pointer"
            >
              <LogOut size={14} aria-hidden="true" />
              <span>ออกจากระบบ</span>
            </button>
          </div>
        </div>
      </header>

      <div className="flex-1 max-w-7xl w-full mx-auto flex flex-col md:flex-row">
        <aside
          className={`${
            sidebarOpen ? 'block' : 'hidden'
          } md:block w-full md:w-64 bg-white text-slate-700 border-r border-rose-100 shrink-0`}
        >
          <div className="p-4 space-y-6">
            <div>
              <p className="px-3 mb-2 text-xs font-medium text-rose-500">
                เมนูการใช้งาน · {activeRole}
              </p>
              <nav className="space-y-1">
                {navItems.map((item) => {
                  const Icon = item.icon;
                  const isActive = activeTab === item.id;
                  return (
                    <button
                      key={item.id}
                      type="button"
                      onClick={() => {
                        setActiveTab(item.id);
                        setSidebarOpen(false);
                      }}
                      className={`w-full flex items-center gap-3 px-3 py-2.5 rounded-lg text-sm font-medium transition-all whitespace-nowrap cursor-pointer ${
                        isActive
                          ? 'bg-gradient-to-r from-pink-500 to-rose-500 text-white shadow-sm shadow-pink-500/20'
                          : 'text-slate-700 hover:bg-rose-50 hover:text-rose-700'
                      }`}
                    >
                      <Icon size={17} className="shrink-0" aria-hidden="true" />
                      <span className="truncate">{item.label}</span>
                    </button>
                  );
                })}
              </nav>
            </div>

            <div className="pt-4 border-t border-rose-100 px-3 space-y-2 text-xs text-slate-500">
              <p className="font-semibold text-rose-900">บัญชีผู้ใช้งาน</p>
              <p className="text-slate-900 font-medium truncate">{currentUser.name}</p>
              {currentUser.employeeId && (
                <div className="flex justify-between font-mono tabular-nums">
                  <span>รหัสพนักงาน</span>
                  <span className="text-pink-700 font-semibold">{currentUser.employeeId}</span>
                </div>
              )}
              <div className="flex justify-between font-mono tabular-nums">
                <span>คำร้องในระบบ</span>
                <span className="text-slate-900 font-medium">{requests.length}</span>
              </div>
              {activeRole === 'Admin' && (
                <div className="flex justify-between font-mono tabular-nums">
                  <span>พนักงานทั้งหมด</span>
                  <span className="text-slate-900 font-medium">{users.length}</span>
                </div>
              )}
            </div>

            {activeRole === 'Customer' && (
              <div className="pt-4 border-t border-rose-100 px-3 space-y-2 text-xs">
                <p className="font-semibold text-rose-900">ติดต่อผู้ดูแลระบบ</p>
                <p className="text-slate-600">
                  LINE ID:{' '}
                  <span className="font-mono font-semibold text-pink-700">
                    {siteConfig.supportLineId}
                  </span>
                </p>
                <p className="text-slate-600">
                  เบอร์ติดต่อ:{' '}
                  <a
                    href={`tel:${siteConfig.supportPhone}`}
                    className="font-mono font-semibold text-pink-700 hover:underline"
                  >
                    {siteConfig.supportPhone}
                  </a>
                </p>
                {siteConfig.supportNote && (
                  <p className="text-slate-500 leading-relaxed">{siteConfig.supportNote}</p>
                )}
                <button
                  type="button"
                  onClick={() => {
                    navigator.clipboard?.writeText(siteConfig.supportLineId);
                    showToast(`คัดลอก LINE ID: ${siteConfig.supportLineId} เรียบร้อยแล้ว`);
                  }}
                  className="w-full py-1.5 px-2.5 bg-rose-50 hover:bg-rose-100 text-rose-700 rounded-md border border-rose-200 text-xs font-medium transition-colors cursor-pointer"
                >
                  คัดลอก LINE ID
                </button>
              </div>
            )}
          </div>
        </aside>

        <main className="flex-1 p-4 sm:p-6 lg:p-8 min-w-0 space-y-6">
          {visibleCustomDetails.length > 0 && activeTab !== 'site-settings' && (
            <div className="space-y-3">
              {visibleCustomDetails.map((detail) => (
                <div
                  key={detail.id}
                  className="bg-white border border-rose-200 rounded-xl p-4 sm:p-5 space-y-1 shadow-xs shadow-pink-500/5"
                >
                  <h2 className="text-sm font-semibold text-rose-950">{detail.title}</h2>
                  <p className="text-xs sm:text-sm text-slate-600 leading-relaxed whitespace-pre-line">
                    {detail.content}
                  </p>
                </div>
              ))}
            </div>
          )}

          {activeRole === 'Customer' &&
            (activeTab === 'dashboard' ||
              activeTab === 'costs' ||
              activeTab === 'user-management' ||
              activeTab === 'site-settings') && (
              <div className="bg-white p-8 rounded-xl border border-rose-200 text-center space-y-4">
                <AlertTriangle size={40} className="mx-auto text-rose-500" aria-hidden="true" />
                <h2 className="text-xl font-bold text-slate-900">
                  ไม่มีสิทธิ์เข้าถึงข้อมูลภาพรวมองค์กร
                </h2>
                <p className="text-sm text-slate-500 max-w-md mx-auto">
                  บัญชีประเภทลูกค้าสามารถใช้งานเฉพาะหน้ายื่นคำขอคืนสินค้าและติดตามสถานะพัสดุคืนเท่านั้น
                </p>
                <button
                  type="button"
                  onClick={() => setActiveTab('new-request')}
                  className="px-4 py-2 bg-pink-600 hover:bg-pink-700 text-white rounded-lg text-sm font-medium cursor-pointer"
                >
                  ไปที่หน้ายื่นคำขอคืนสินค้า
                </button>
              </div>
            )}

          {activeTab === 'dashboard' && activeRole === 'Admin' && (
            <DashboardView
              requests={requests}
              notifications={notifications}
              onExportCSV={handleExportCSV}
              onSelectRequest={(req, targetTab) => {
                setSelectedRequestId(req.id);
                if (targetTab) setActiveTab(targetTab);
              }}
              onNavigateTab={setActiveTab}
            />
          )}

          {activeTab === 'requests' && (
            <RequestsView
              requests={requests}
              activeRole={activeRole}
              searchTerm={searchTerm}
              onSearchChange={setSearchTerm}
              filterStatus={filterStatus}
              onFilterStatusChange={setFilterStatus}
              onUpdateStatus={handleUpdateStatus}
              onReceiveWarehouse={handleReceiveWarehouse}
              onGoToTracking={(req) => {
                setSelectedRequestId(req.id);
                setActiveTab('tracking');
              }}
              onInspectDetails={(req) => setModalRequest(req)}
              onNavigateTab={setActiveTab}
            />
          )}

          {activeTab === 'new-request' && (
            <NewRequestView
              products={products}
              returnReasons={returnReasons}
              siteConfig={siteConfig}
              onSubmitRequest={handleCreateRequest}
            />
          )}

          {activeTab === 'tracking' && (
            <TrackingView
              requests={requests}
              selectedRequest={selectedRequest}
              onSelectRequest={(req) => setSelectedRequestId(req ? req.id : '')}
              activeRole={activeRole}
              siteConfig={siteConfig}
              onUpdateStatus={handleUpdateStatus}
              onReceiveWarehouse={handleReceiveWarehouse}
              onSaveCustomerReturnTracking={handleSaveCustomerReturnTracking}
              onSaveCustomerEmail={handleSaveCustomerEmail}
              onSendRefundSlip={handleSendRefundSlip}
            />
          )}

          {activeTab === 'warehouse' &&
            (activeRole === 'Admin' || activeRole === 'Warehouse Staff') && (
              <WarehouseView
                requests={requests}
                warehouses={warehouses}
                products={products}
                activeRole={activeRole}
                onUpdateStatus={handleUpdateStatus}
                onReceiveWarehouse={handleReceiveWarehouse}
                onDispatchToCustomer={handleDispatchToCustomer}
                onSendRefundSlip={handleSendRefundSlip}
              />
            )}

          {activeTab === 'inspection' &&
            (activeRole === 'Admin' || activeRole === 'Inspector') && (
              <InspectionView
                requests={requests}
                activeRole={activeRole}
                onUpdateStatus={handleUpdateStatus}
                onCompleteInspection={handleCompleteInspection}
              />
            )}

          {activeTab === 'costs' && activeRole === 'Admin' && (
            <CostManagementView
              requests={requests}
              onUpdateCost={handleUpdateCost}
              onExportCSV={handleExportCSV}
            />
          )}

          {activeTab === 'user-management' &&
            activeRole === 'Admin' &&
            currentUser.level === 1 && (
              <UserManagementView
                users={users}
                onCreateUser={handleCreateUser}
                onDeleteUser={handleDeleteUser}
              />
            )}

          {activeTab === 'site-settings' &&
            activeRole === 'Admin' &&
            currentUser.level === 1 && (
              <SiteSettingsView
                key={`${siteConfig.brandName}-${siteConfig.supportLineId}-${siteConfig.customDetails.length}`}
                siteConfig={siteConfig}
                products={products}
                returnReasons={returnReasons}
                onSaveSiteConfig={handleSaveSiteConfig}
                onResetSiteConfig={handleResetSiteConfig}
                onAddProduct={handleAddProduct}
                onDeleteProduct={handleDeleteProduct}
                onUpdateReturnReasons={handleUpdateReturnReasons}
              />
            )}
        </main>
      </div>

      {pendingVerify && (
        <StaffVerifyModal
          actionLabel={pendingVerify.actionLabel}
          currentUser={currentUser}
          users={users}
          onCancel={() => setPendingVerify(null)}
          onVerified={(verifiedUser) => {
            const confirmFn = pendingVerify.onConfirm;
            setPendingVerify(null);
            confirmFn(verifiedUser);
          }}
        />
      )}

      {modalRequest && (
        <div
          className="fixed inset-0 z-50 bg-rose-950/40 backdrop-blur-xs flex items-center justify-center p-4"
          role="dialog"
          aria-modal="true"
          aria-labelledby="modal-title"
        >
          <div className="bg-white border border-rose-200 rounded-xl max-w-2xl w-full max-h-[90vh] overflow-y-auto p-6 space-y-5 shadow-xl shadow-pink-500/10">
            <div className="flex items-start justify-between gap-4 border-b border-rose-100 pb-4">
              <div>
                <h2 id="modal-title" className="text-lg font-bold font-mono text-slate-900">
                  {modalRequest.id}
                </h2>
                <p className="text-xs text-slate-500 font-mono mt-0.5">
                  คำสั่งซื้อ {modalRequest.orderNumber} · ยื่นคำร้อง {modalRequest.requestDate}
                </p>
              </div>
              <button
                type="button"
                onClick={() => setModalRequest(null)}
                aria-label="ปิดหน้าต่างรายละเอียด"
                className="p-1.5 text-rose-400 hover:text-rose-700 hover:bg-rose-50 rounded-lg cursor-pointer"
              >
                <X size={20} />
              </button>
            </div>

            <div className="grid grid-cols-1 sm:grid-cols-3 gap-5">
              <div className="aspect-[4/3] sm:aspect-square rounded-lg overflow-hidden border border-rose-200 bg-rose-50">
                <ResilientImage src={modalRequest.imageUrl} alt={modalRequest.productName} />
              </div>
              <div className="sm:col-span-2 space-y-3 text-sm">
                <StatusIndicator status={modalRequest.status} />
                <div>
                  <p className="font-semibold text-slate-900">{modalRequest.productName}</p>
                  <p className="text-xs font-mono text-slate-500">
                    {modalRequest.sku} · จำนวน {modalRequest.qty} ชิ้น ·{' '}
                    {formatCurrency(modalRequest.unitPrice * modalRequest.qty)}
                  </p>
                </div>
                <div className="text-xs text-slate-600 space-y-1">
                  <p>
                    <span className="font-medium text-slate-800">ลูกค้า:</span>{' '}
                    {modalRequest.customerName} ({modalRequest.customerPhone})
                  </p>
                  <p>
                    <span className="font-medium text-slate-800">สาเหตุ:</span>{' '}
                    {modalRequest.reason}
                  </p>
                  <p className="leading-relaxed">{modalRequest.detail}</p>
                </div>
                <div className="pt-2 border-t border-rose-100 text-xs font-mono text-slate-600 flex flex-wrap gap-x-4 gap-y-1">
                  <span>คลัง: {modalRequest.warehouseId}</span>
                  <span>Bin: {modalRequest.locationBin}</span>
                  <span>ต้นทุนรวม: {formatCurrency(getTotalRequestCost(modalRequest))}</span>
                </div>
              </div>
            </div>

            {modalRequest.resolution && (
              <div className="p-3.5 rounded-lg bg-rose-50/70 border border-rose-200 text-xs space-y-1">
                <p className="font-semibold text-rose-950">
                  ความประสงค์ของลูกค้า:{' '}
                  {modalRequest.resolution.type === 'reship'
                    ? 'ต้องการจัดส่งสินค้าให้ใหม่'
                    : 'ต้องการขอรับเงินคืน'}
                </p>
                {modalRequest.resolution.type === 'reship' ? (
                  <p className="text-slate-600">
                    ผู้รับ: {modalRequest.resolution.reshipName} · โทร{' '}
                    <span className="font-mono">{modalRequest.resolution.reshipPhone}</span>
                    <br />
                    ที่อยู่จัดส่ง: {modalRequest.resolution.reshipAddress}
                  </p>
                ) : (
                  <p className="text-slate-600">
                    {modalRequest.resolution.refundBankName} · ชื่อบัญชี:{' '}
                    {modalRequest.resolution.refundAccountName} · เลขบัญชี:{' '}
                    <span className="font-mono font-semibold text-slate-900">
                      {modalRequest.resolution.refundAccountNumber}
                    </span>
                  </p>
                )}
              </div>
            )}

            {modalRequest.customerReturnTracking &&
              modalRequest.resolution?.type !== 'refund' && (
                <div className="p-3.5 rounded-lg bg-rose-50/70 border border-rose-200 text-xs space-y-1">
                  <p className="font-semibold text-rose-950">
                    เลขพัสดุที่ลูกค้าส่งคืน ({modalRequest.customerReturnCarrier || 'ขนส่งทั่วไป'})
                  </p>
                  <p className="font-mono font-bold text-pink-700">
                    {modalRequest.customerReturnTracking}
                  </p>
                </div>
              )}

            {modalRequest.refundSlip && (
              <div className="p-3.5 rounded-lg bg-emerald-50 border border-emerald-200 text-xs space-y-1.5">
                <p className="font-semibold text-emerald-950">
                  ส่งสลิปโอนเงินคืนถึงอีเมลลูกค้าแล้ว
                </p>
                <p className="text-slate-700">
                  อีเมลผู้รับ:{' '}
                  <span className="font-mono font-semibold text-slate-900">
                    {modalRequest.refundSlip.sentToEmail}
                  </span>{' '}
                  · ยอดเงินคืน:{' '}
                  <span className="font-mono font-bold text-emerald-700">
                    {formatCurrency(modalRequest.refundSlip.amount)}
                  </span>{' '}
                  · อ้างอิง:{' '}
                  <span className="font-mono font-semibold text-slate-900">
                    {modalRequest.refundSlip.transferRef}
                  </span>
                </p>
                <p className="text-slate-500 font-mono">
                  ส่งเมื่อ {modalRequest.refundSlip.sentAt} โดย {modalRequest.refundSlip.sentBy}
                </p>
              </div>
            )}

            {modalRequest.outboundDispatch && (
              <div className="p-3.5 rounded-lg bg-pink-50 border border-pink-200 text-xs space-y-1">
                <p className="font-semibold text-pink-950">
                  ข้อมูลการจัดส่งพัสดุถึงลูกค้า ({modalRequest.outboundDispatch.carrier})
                </p>
                <p className="text-slate-700">
                  เลขที่พัสดุ:{' '}
                  <span className="font-mono font-bold text-pink-700">
                    {modalRequest.outboundDispatch.trackingNumber}
                  </span>{' '}
                  · สินค้า: <span className="font-semibold">{modalRequest.outboundDispatch.productName}</span>
                </p>
                <p className="text-slate-500 font-mono">
                  จัดส่งเมื่อ {modalRequest.outboundDispatch.shippedAt} โดย{' '}
                  {modalRequest.outboundDispatch.shippedBy}
                </p>
              </div>
            )}

            {modalRequest.timeline.length > 0 && (
              <div className="pt-4 border-t border-rose-100 space-y-2">
                <p className="text-xs font-semibold text-slate-800">ประวัติการดำเนินการ</p>
                <div className="divide-y divide-rose-100 border border-rose-200 rounded-lg bg-rose-50/50">
                  {modalRequest.timeline.map((t, i) => (
                    <div key={i} className="px-3.5 py-2 flex items-center justify-between gap-3 text-xs">
                      <div>
                        <span className="font-medium text-slate-900">{t.status}</span>
                        <span className="text-slate-500 ml-2">โดย {t.actor} — {t.note}</span>
                      </div>
                      <span className="font-mono text-slate-500 shrink-0">{t.date}</span>
                    </div>
                  ))}
                </div>
              </div>
            )}

            <div className="flex flex-wrap items-center justify-end gap-2 pt-4 border-t border-rose-100">
              {activeRole === 'Admin' && (
                <button
                  type="button"
                  onClick={() => {
                    setSelectedRequestId(modalRequest.id);
                    setActiveTab('tracking');
                    setModalRequest(null);
                  }}
                  className="px-4 py-2 bg-pink-600 hover:bg-pink-700 text-white text-xs font-semibold rounded-lg cursor-pointer"
                >
                  ติดตามสถานะ 7 ขั้นตอน
                </button>
              )}
              <button
                type="button"
                onClick={() => setModalRequest(null)}
                className="px-4 py-2 bg-rose-50 hover:bg-rose-100 text-rose-700 text-xs font-medium rounded-lg cursor-pointer"
              >
                ปิดหน้าต่าง
              </button>
            </div>
          </div>
        </div>
      )}

      {toast && (
        <div
          role="status"
          className={`fixed bottom-5 right-5 z-50 px-4 py-3 rounded-lg shadow-lg border text-sm font-medium flex items-center gap-2 transition-opacity ${
            toast.type === 'success'
              ? 'bg-white text-rose-950 border-pink-300 shadow-pink-500/15'
              : 'bg-rose-950 text-rose-50 border-rose-800'
          }`}
        >
          <CheckCircle2
            size={16}
            className={toast.type === 'success' ? 'text-pink-600 shrink-0' : 'text-pink-300 shrink-0'}
            aria-hidden="true"
          />
          <span>{toast.msg}</span>
        </div>
      )}
    </div>
  );
}
